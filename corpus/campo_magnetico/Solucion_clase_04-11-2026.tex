\section*{Problema: Fuerza entre tres alambres paralelos (Gu\'ia 9, Problema 2)}

\noindent Tres alambres rectos, paralelos y muy largos transportan cada uno una corriente
$I = 4\ [\text{A}]$ y est\'an separados una distancia $d = 15\ [\text{cm}]$ entre alambres vecinos,
como muestra la Figura 1. Los alambres extremos (1) y (3) conducen la corriente hacia la derecha
y el alambre central (2) hacia la izquierda. Calcule la \textbf{fuerza magn\'etica por unidad de
longitud} sobre cada alambre.

\medskip

\noindent
\begin{minipage}[t]{0.46\textwidth}
\centering
\vspace{0pt}

    \draw[<->,gray] (0.2,0)--(0.2,-2); \node[right,font=\footnotesize,gray] at (0.2,-1){$d$};
    % ---- sistema de referencia ----
    \ejesxyz{2.2,-3.3}
\end{tikzpicture}\\[2pt]
{\footnotesize Figura 1}
\end{minipage}%
\hfill
\begin{minipage}[t]{0.50\textwidth}
\vspace{0pt}
\textbf{Datos:}
\[
    I = 4\ [\text{A}],\qquad d = 15\ [\text{cm}] = 0{,}15\ [\text{m}]
\]
\[
    \mu_0 = 4\pi\times10^{-7}\ [\text{T}\cdot\text{m/A}]
\]

\smallskip
\textbf{Sistema de referencia} (cl\'asico):
\begin{itemize}\setlength\itemsep{1pt}
    \item eje $x^{+}$ hacia la derecha ($\hat{\imath}$),
    \item eje $y^{+}$ hacia arriba ($\hat{\jmath}$),
    \item eje $z^{+}$ saliendo de la p\'agina ($\hat{k}$).
\end{itemize}
Los tres alambres est\'an en el plano $xy$, paralelos al eje $x$, en
$y_1 = +d$, $y_2 = 0$ y $y_3 = -d$.
\end{minipage}

\medskip

\noindent Con este sistema de referencia, el vector longitud $\vec{L}$ de cada alambre (un tramo de
largo $l$ orientado seg\'un el sentido de su corriente) es:
\[
    \vec{L}_1 = l\,\hat{\imath},\qquad
    \vec{L}_2 = -\,l\,\hat{\imath},\qquad
    \vec{L}_3 = l\,\hat{\imath}
\]

\noindent\textbf{Estrategia:} (1) obtener con la ley de Amp\`ere la magnitud del campo de un alambre
recto infinito; (2) obtener su direcci\'on con la regla de la mano derecha y expresarla con puntos y
cruces; (3) escribir el campo que cada alambre produce en la posici\'on de los otros dos; (4) aplicar
$\vec{F} = I\,\vec{L}\times\vec{B}$ a cada par de alambres y sumar.

% 1. LEY DE AMPERE
%========================================================
\section*{1.\ Campo magn\'etico de un alambre recto infinito: ley de Amp\`ere}

\noindent
\begin{minipage}[t]{0.40\textwidth}
\centering
\vspace{0pt}
\begin{tikzpicture}[scale=1.0, >=Stealth, line width=0.9pt]

    % alambre
    \draw[orange!75!black,line width=2.4pt] (0,-2.0)--(0,2.2);
    \draw[->,violet,line width=1.4pt] (0.25,1.1)--(0.25,2.0);

    \draw[->,blue,line width=1.5pt] (0.75,-0.39)--(1.55,-0.30);
    \node[blue,right,font=\small] at (1.55,-0.30){$\vec{B}$};
    \draw[->,red!70!black,line width=1.1pt] (0.75,-0.53)--(1.25,-0.49);
    \node[red!70!black,below,font=\small] at (1.10,-0.55){$d\vec{\ell}$};
    \node[red,font=\footnotesize,align=center] at (-1.25,0.75){trayectoria\\amperiana};
\end{tikzpicture}\\[2pt]
{\footnotesize Figura 2}
\end{minipage}%
\hfill
\begin{minipage}[t]{0.56\textwidth}
\vspace{0pt}
\textbf{Ley de Amp\`ere:}
\[
    \oint \vec{B}\cdot d\vec{\ell} = \mu_0\, I_{\text{enc}}
\]
\textbf{Paso 1 -- Trayectoria amperiana.} Por la simetr\'ia cil\'indrica del alambre infinito,
se elige una circunferencia de radio $r$ centrada en el alambre y perpendicular a \'el
(Figura 2, l\'inea roja segmentada).

\smallskip
\textbf{Paso 2 -- Simetr\'ia.} Sobre esa circunferencia:
\begin{itemize}\setlength\itemsep{1pt}
    \item $|\vec{B}|$ es el mismo en todos los puntos (todos est\'an a la misma distancia $r$);
    \item $\vec{B}$ es tangente a la circunferencia, es decir, paralelo a $d\vec{\ell}$:
          $\vec{B}\cdot d\vec{\ell} = B\,d\ell$.
\end{itemize}
\end{minipage}

\medskip

\noindent\textbf{Paso 3 -- Circulaci\'on.} Como $B$ es constante sobre la trayectoria, sale de la
integral:
\[
    \oint \vec{B}\cdot d\vec{\ell} = \oint B\,d\ell = B\oint d\ell = B\,(2\pi r)
\]

\noindent\textbf{Paso 4 -- Corriente encerrada.} La circunferencia encierra todo el alambre:
$I_{\text{enc}} = I$.

\medskip
\noindent\textbf{Paso 5 -- Despeje.} Igualando ambos lados de la ley de Amp\`ere:
\[
    B\,(2\pi r) = \mu_0 I
    \qquad\Longrightarrow\qquad
    \boxed{B(r) = \frac{\mu_0\, I}{2\pi r}}
\]
El campo decae como $1/r$ y sus l\'ineas son circunferencias conc\'entricas al alambre.

%========================================================
% 2. REGLA DE LA MANO DERECHA
%========================================================
\section*{2.\ Direcci\'on del campo: regla de la mano derecha}

\noindent La ley de Amp\`ere entrega la magnitud de $\vec{B}$; su \textbf{sentido} se obtiene con la
\textbf{regla de la mano derecha}: se toma el alambre con la mano derecha, con el
\textbf{pulgar apuntando en el sentido de la corriente} $I$; los \textbf{dedos que envuelven el
alambre} indican el sentido en que circulan las l\'ineas de $\vec{B}$.

\noindent\textbf{Forma vectorial equivalente.} Si $\hat{u}_I$ es el unitario en el sentido de la
corriente y $\hat{u}_r$ el unitario perpendicular que va \textbf{desde el alambre hacia el punto}
donde se calcula el campo, la regla de la mano derecha equivale a:
\[
    \hat{u}_B = \hat{u}_I\times\hat{u}_r
    \qquad\Longrightarrow\qquad
    \vec{B} = \frac{\mu_0 I}{2\pi r}\,\bigl(\hat{u}_I\times\hat{u}_r\bigr)
\]
Esta forma permite verificar con productos cruz de unitarios lo que indica la mano derecha.

% 3. PUNTOS Y CRUCES
%========================================================
\section*{3.\ Notaci\'on de puntos y cruces}

\noindent Cuando la corriente circula \emph{en el plano de la p\'agina}, el campo que produce en ese
mismo plano es perpendicular a la p\'agina. Se representa con:
\begin{center}
\begin{tabular}{c@{\qquad}l}
\begin{tikzpicture}[baseline=-3pt]\punto{0,0}\end{tikzpicture} & punto: $\vec{B}$
      \textbf{sale} de la p\'agina \;$\Rightarrow\; +\hat{k}$ \quad(punta de la flecha)\\[4pt]
\begin{tikzpicture}[baseline=-3pt]\cruz{0,0}\end{tikzpicture} & cruz: $\vec{B}$
      \textbf{entra} a la p\'agina \;$\Rightarrow\; -\hat{k}$ \quad(cola de la flecha)
\end{tabular}
\end{center}

\noindent Para un alambre horizontal, la mano derecha (pulgar seg\'un $I$) indica que los dedos
\emph{salen} de la p\'agina por un lado del alambre y \emph{entran} por el otro:

\begin{center}
\begin{tikzpicture}[scale=0.9, >=Stealth, line width=0.9pt]

\noindent Verificaci\'on con la forma vectorial: para $I$ hacia $+x$, en un punto de arriba
$\hat{\imath}\times\hat{\jmath} = \hat{k}$ (punto) y en un punto de abajo
$\hat{\imath}\times(-\hat{\jmath}) = -\hat{k}$ (cruz). Para $I$ hacia $-x$ los signos se invierten.

\begin{align*}

    B_{2d} &= \frac{\mu_0 I}{2\pi (2d)}
            = \frac{(4\pi\times10^{-7})(4)}{2\pi(0{,}30)}
            = 2{,}667\times10^{-6}\ [\text{T}]
            &&\text{(alambres extremos, } r = 2d\text{)}
\end{align*}

\section*{5.\ Fuerza magn\'etica sobre cada alambre}

\noindent La fuerza sobre un tramo recto $\vec{L}$ con corriente $I$ en un campo $\vec{B}$ es
\[
    \vec{F} = I\,\vec{L}\times\vec{B}
    \qquad\Longrightarrow\qquad
    \frac{\vec{F}}{l} = I\,\frac{\vec{L}}{l}\times\vec{B}
\]
Se calcula la fuerza de \textbf{cada} alambre vecino por separado y luego se suman.

%--------------------------------------------------
\subsection*{Alambre superior (1):\; $\vec{L}_1 = l\,\hat{\imath}$}

\noindent\textbf{Fuerza de (2) sobre (1):}
\begin{align*}
    \vec{F}_{2\to1} &= I\,\vec{L}_1\times\vec{B}_{2\to1}
                     = I\,(l\,\hat{\imath})\times(-B_d\,\hat{k})\\[3pt]
                    &= -\,I\,l\,B_d\,(\hat{\imath}\times\hat{k})
                     = -\,I\,l\,B_d\,(-\hat{\jmath})
                     = I\,l\,B_d\,\hat{\jmath}\\[3pt]
    \frac{\vec{F}_{2\to1}}{l} &= I\,B_d\,\hat{\jmath}
                     = (4)(5{,}333\times10^{-6})\,\hat{\jmath}
                     = 2{,}133\times10^{-5}\,\hat{\jmath}\ [\text{N/m}]
\end{align*}

\noindent\textbf{Fuerza de (3) sobre (1):}
\begin{align*}
    \vec{F}_{3\to1} &= I\,\vec{L}_1\times\vec{B}_{3\to1}
                     = I\,(l\,\hat{\imath})\times(B_{2d}\,\hat{k})\\[3pt]
                    &= I\,l\,B_{2d}\,(\hat{\imath}\times\hat{k})
                     = I\,l\,B_{2d}\,(-\hat{\jmath})
                     = -\,I\,l\,B_{2d}\,\hat{\jmath}\\[3pt]
    \frac{\vec{F}_{3\to1}}{l} &= -\,I\,B_{2d}\,\hat{\jmath}
                     = -(4)(2{,}667\times10^{-6})\,\hat{\jmath}
                     = -1{,}067\times10^{-5}\,\hat{\jmath}\ [\text{N/m}]
\end{align*}

\noindent\textbf{Fuerza neta sobre (1):}
\begin{align*}
    \frac{\vec{F}_1}{l} &= \frac{\vec{F}_{2\to1}}{l} + \frac{\vec{F}_{3\to1}}{l}
                         = (2{,}133 - 1{,}067)\times10^{-5}\,\hat{\jmath}
\end{align*}
\[
    \boxed{\frac{\vec{F}_1}{l} = 1{,}07\times10^{-5}\,\hat{\jmath}\ [\text{N/m}]
    \qquad\text{(hacia arriba)}}
\]

%--------------------------------------------------
\subsection*{Alambre central (2):\; $\vec{L}_2 = -\,l\,\hat{\imath}$}

\noindent\textbf{Fuerza de (1) sobre (2):}
\begin{align*}
    \vec{F}_{1\to2} &= I\,\vec{L}_2\times\vec{B}_{1\to2}
                     = I\,(-\,l\,\hat{\imath})\times(-B_d\,\hat{k})\\[3pt]
                    &= I\,l\,B_d\,(\hat{\imath}\times\hat{k})
                     = I\,l\,B_d\,(-\hat{\jmath})
                     = -\,I\,l\,B_d\,\hat{\jmath}\\[3pt]
    \frac{\vec{F}_{1\to2}}{l} &= -\,I\,B_d\,\hat{\jmath}
                     = -(4)(5{,}333\times10^{-6})\,\hat{\jmath}
                     = -2{,}133\times10^{-5}\,\hat{\jmath}\ [\text{N/m}]
\end{align*}

\noindent\textbf{Fuerza de (3) sobre (2):}
\begin{align*}
    \vec{F}_{3\to2} &= I\,\vec{L}_2\times\vec{B}_{3\to2}
                     = I\,(-\,l\,\hat{\imath})\times(B_d\,\hat{k})\\[3pt]
                    &= -\,I\,l\,B_d\,(\hat{\imath}\times\hat{k})
                     = -\,I\,l\,B_d\,(-\hat{\jmath})
                     = I\,l\,B_d\,\hat{\jmath}\\[3pt]
    \frac{\vec{F}_{3\to2}}{l} &= I\,B_d\,\hat{\jmath}
                     = (4)(5{,}333\times10^{-6})\,\hat{\jmath}
                     = 2{,}133\times10^{-5}\,\hat{\jmath}\ [\text{N/m}]
\end{align*}

\noindent\textbf{Fuerza neta sobre (2):}
\begin{align*}
    \frac{\vec{F}_2}{l} &= \frac{\vec{F}_{1\to2}}{l} + \frac{\vec{F}_{3\to2}}{l}
                         = (-2{,}133 + 2{,}133)\times10^{-5}\,\hat{\jmath}
\end{align*}
\[
    \boxed{\frac{\vec{F}_2}{l} = \vec{0}\ [\text{N/m}]}
\]
\noindent El resultado es consistente con el campo total en la posici\'on de (2):
$\vec{B}_{1\to2}+\vec{B}_{3\to2} = -B_d\,\hat{k} + B_d\,\hat{k} = \vec{0}$. Las dos fuerzas existen
y son de igual magnitud, pero de sentidos opuestos: (1) empuja a (2) hacia abajo y (3) lo empuja
hacia arriba.

%--------------------------------------------------
\subsection*{Alambre inferior (3):\; $\vec{L}_3 = l\,\hat{\imath}$}

\noindent\textbf{Fuerza de (1) sobre (3):}
\begin{align*}
    \vec{F}_{1\to3} &= I\,\vec{L}_3\times\vec{B}_{1\to3}
                     = I\,(l\,\hat{\imath})\times(-B_{2d}\,\hat{k})\\[3pt]
                    &= -\,I\,l\,B_{2d}\,(\hat{\imath}\times\hat{k})
                     = -\,I\,l\,B_{2d}\,(-\hat{\jmath})
                     = I\,l\,B_{2d}\,\hat{\jmath}\\[3pt]
    \frac{\vec{F}_{1\to3}}{l} &= I\,B_{2d}\,\hat{\jmath}
                     = (4)(2{,}667\times10^{-6})\,\hat{\jmath}
                     = 1{,}067\times10^{-5}\,\hat{\jmath}\ [\text{N/m}]
\end{align*}

\noindent\textbf{Fuerza de (2) sobre (3):}
\begin{align*}
    \vec{F}_{2\to3} &= I\,\vec{L}_3\times\vec{B}_{2\to3}
                     = I\,(l\,\hat{\imath})\times(B_d\,\hat{k})\\[3pt]
                    &= I\,l\,B_d\,(\hat{\imath}\times\hat{k})
                     = I\,l\,B_d\,(-\hat{\jmath})
                     = -\,I\,l\,B_d\,\hat{\jmath}\\[3pt]
    \frac{\vec{F}_{2\to3}}{l} &= -\,I\,B_d\,\hat{\jmath}
                     = -(4)(5{,}333\times10^{-6})\,\hat{\jmath}
                     = -2{,}133\times10^{-5}\,\hat{\jmath}\ [\text{N/m}]
\end{align*}

\noindent\textbf{Fuerza neta sobre (3):}
\begin{align*}
    \frac{\vec{F}_3}{l} &= \frac{\vec{F}_{1\to3}}{l} + \frac{\vec{F}_{2\to3}}{l}
                         = (1{,}067 - 2{,}133)\times10^{-5}\,\hat{\jmath}
\end{align*}
\[
    \boxed{\frac{\vec{F}_3}{l} = -1{,}07\times10^{-5}\,\hat{\jmath}\ [\text{N/m}]
    \qquad\text{(hacia abajo)}}
\]

%--------------------------------------------------
\subsection*{Verificaci\'on: tercera ley de Newton}

\noindent Las fuerzas entre cada par de alambres son de igual magnitud y sentido opuesto:
\begin{align*}
    \vec{F}_{2\to1} &= +2{,}133\times10^{-5}\,\hat{\jmath} = -\,\vec{F}_{1\to2},\\
    \vec{F}_{3\to1} &= -1{,}067\times10^{-5}\,\hat{\jmath} = -\,\vec{F}_{1\to3},\\
    \vec{F}_{3\to2} &= +2{,}133\times10^{-5}\,\hat{\jmath} = -\,\vec{F}_{2\to3}
    \qquad [\text{N/m}]
\end{align*}
Por lo tanto, la suma de las fuerzas sobre los tres alambres es nula:
\[
    \frac{\vec{F}_1}{l} + \frac{\vec{F}_2}{l} + \frac{\vec{F}_3}{l}
    = (1{,}067 + 0 - 1{,}067)\times10^{-5}\,\hat{\jmath} = \vec{0}
\]

\noindent\textbf{Interpretaci\'on:} del c\'alculo vectorial se desprende la regla conocida:

del vecino a distancia $d$ es el doble de la atracci\'on del alambre lejano a distancia $2d$.
