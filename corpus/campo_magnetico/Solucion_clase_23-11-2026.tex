\section*{Problema 1: Campo de dos arcos en planos perpendiculares \textcolor{red}{(22 ptos)}}

\noindent En el sistema de coordenadas cartesiano de la figura se tienen dos filamentos conductores
circulares, ambos centrados en el origen, que transportan una corriente constante $I$. El primer
filamento es un arco de radio $R_1$ en el plano $xz$ que forma un semic\'irculo (\'angulo total
$\pi$), por el cual circula la corriente en sentido antihorario. El segundo filamento es un arco de
radio $R_2$ en el plano $xy$ que abarca un \'angulo $\alpha = \pi/4$, por el cual la corriente
tambi\'en circula en sentido antihorario. Usando la ley de Biot-Savart, determine: (a) el campo en el
origen debido al filamento $R_1$; (b) el campo en el origen debido al filamento $R_2$; (c) el campo
total en el origen. Desprecie la contribuci\'on de los segmentos rectos.

\medskip

\noindent

\centering

\begin{minipage}[c]{0.50\textwidth}
\textbf{Campo de un elemento de arco.} Para un elemento $d\ell = R\,d\varphi$ de un arco de radio
$R$ centrado en el origen, el vector $d\vec{\ell}$ es siempre perpendicular al radio que lo une con
el origen ($\sin90^\circ = 1$) y la distancia es constante e igual a $R$:
\[
    dB = \frac{\mu_0 I}{4\pi}\,\frac{d\ell\,\sin 90^\circ}{R^2}
       = \frac{\mu_0 I}{4\pi}\,\frac{R\,d\varphi}{R^2}
       = \frac{\mu_0 I}{4\pi R}\,d\varphi
\]
\textbf{Geometr\'ia (seg\'un la figura):}
\begin{itemize}\setlength\itemsep{1pt}
    \item Arco $R_1$: comienza en el eje $+z$, pasa por $+x$ y termina en $-z$ (plano $xz$).
    \item Arco $R_2$: comienza en el eje $+y$ y avanza hacia $-x$ un \'angulo $\alpha = \pi/4$
          (plano $xy$).
\end{itemize}
\end{minipage}

\medskip
\noindent Como todos los elementos de un mismo arco producen en el origen un $d\vec{B}$ con la
\textbf{misma direcci\'on} (perpendicular al plano del arco), basta integrar el m\'odulo sobre el
\'angulo que abarca el arco y luego asignar la direcci\'on con la regla de la mano derecha.

%--------------------------------------------------
\subsection*{(a) Campo del filamento de radio $R_1$ \ptos{10}}

\noindent\textbf{M\'odulo.} El semic\'irculo abarca un \'angulo $\pi$:
\[
    B_1 = \int_0^{\pi}\frac{\mu_0 I}{4\pi R_1}\,d\varphi
        = \frac{\mu_0 I}{4\pi R_1}\,(\pi - 0)
        = \frac{\mu_0 I}{4R_1}
\]

\noindent\textbf{Direcci\'on (regla de la mano derecha).} Los dedos de la mano derecha se curvan en
el sentido de la corriente, que recorre el arco $+z\to+x\to-z$ en el plano $xz$ (antihorario visto
desde $+y$). El pulgar apunta en la direcci\'on de $\vec{B}$ en el centro del arco: \textbf{$+y$}.
(Equivalentemente, girar de $\hat{k}$ hacia $\hat{\imath}$ corresponde a
$\hat{k}\times\hat{\imath} = \hat{\jmath}$.)
\[
    \boxed{\vec{B}_1 = \frac{\mu_0 I}{4R_1}\,\hat{\jmath}}
\]

%--------------------------------------------------
\subsection*{(b) Campo del filamento de radio $R_2$ \ptos{10}}

\noindent\textbf{M\'odulo.} El arco abarca un \'angulo $\alpha = \pi/4$:
\[
    B_2 = \int_0^{\pi/4}\frac{\mu_0 I}{4\pi R_2}\,d\varphi
        = \frac{\mu_0 I}{4\pi R_2}\cdot\frac{\pi}{4}
        = \frac{\mu_0 I}{16R_2}
\]

\noindent\textbf{Direcci\'on (regla de la mano derecha).} La corriente recorre el arco desde $+y$
hacia $-x$ en el plano $xy$, es decir, en sentido antihorario visto desde $+z$. Con los dedos en ese
sentido, el pulgar apunta en \textbf{$+z$}. (Equivalentemente, girar de $\hat{\jmath}$ hacia
$-\hat{\imath}$ corresponde a $\hat{\jmath}\times(-\hat{\imath}) = \hat{k}$.)
\[
    \boxed{\vec{B}_2 = \frac{\mu_0 I}{16R_2}\,\hat{k}}
\]

%--------------------------------------------------
\subsection*{(c) Campo total en el origen \ptos{2}}

\noindent Por superposici\'on:
\[
    \boxed{\vec{B} = \vec{B}_1 + \vec{B}_2 = \frac{\mu_0 I}{4R_1}\,\hat{\jmath} + \frac{\mu_0 I}{16R_2}\,\hat{k}}
    \qquad\Longrightarrow\qquad
    |\vec{B}| = \mu_0 I\sqrt{\frac{1}{256\,R_2^2} + \frac{1}{16\,R_1^2}}
\]

\section*{Problema 2: Fuerza entre tres alambres paralelos \textcolor{red}{(22 ptos)}}

\noindent Se dispone de un sistema de tres alambres muy largos por donde circulan corrientes
uniformes y paralelas entre s\'i (radios despreciables). La figura muestra la vista de perfil. Las
corrientes (2) y (3) est\'an a una distancia $d$ de la corriente (1), con $\theta = 60^\circ$. La
corriente (1) \textbf{sale} del plano; las corrientes (2) y (3) \textbf{entran} a \'el.
(a) Calcule con la ley de Amp\`ere el campo que producen las corrientes (2) y (3). (b) Para una
secci\'on de longitud $L$ del alambre (1), calcule la fuerza magn\'etica debida a los campos de (2)
y (3). (c) Si $I_1 = 1$ [A], $I_2 = I_3 = 2$ [A] y $d = 2L = 20$ [cm], determine la fuerza neta sobre
la corriente (1), en funci\'on de $\mu_0$ y $\pi$.

\subsection*{(a) Campo de las corrientes (2) y (3): ley de Amp\`ere \ptos{6}}

\noindent Se elige una trayectoria amperiana circular de radio $r$ centrada en el alambre. Por
simetr\'ia, $\vec{B}$ es tangente a ella y de m\'odulo constante:
\[
    \oint\vec{B}\cdot d\vec{\ell} = B\,(2\pi r) = \mu_0 I
    \qquad\Longrightarrow\qquad
    B_2 = \frac{\mu_0 I_2}{2\pi r},\qquad B_3 = \frac{\mu_0 I_3}{2\pi r}
\]
\noindent\textbf{Direcci\'on (regla de la mano derecha):} como (2) y (3) \textbf{entran} al plano,
con el pulgar hacia la hoja los dedos giran en sentido \textbf{horario}. En la posici\'on de (1),
a distancia $r = d$ de cada alambre, el campo es tangente a la circunferencia centrada en el alambre
fuente, es decir, perpendicular a la l\'inea radial $\hat{u}_r$ que une (1) con ese alambre:

\subsection*{(b) Fuerza sobre una secci\'on $L$ del alambre (1) \ptos{10}}

\noindent Como la corriente (1) sale del plano, $\vec{L} = L\,\hat{k}$. Con
$\vec{F} = I_1\,\vec{L}\times\vec{B}$:
\begin{align*}
    \vec{F}_{2\to1} &= I_1 L\,\hat{k}\times\frac{\mu_0 I_2}{2\pi d}
                       \Bigl(-\tfrac{\sqrt{3}}{2}\,\hat{\imath} + \tfrac{1}{2}\,\hat{\jmath}\Bigr)\\[3pt]
    &= \frac{\mu_0 I_1 I_2 L}{2\pi d}
       \Bigl[-\tfrac{\sqrt{3}}{2}\,(\hat{k}\times\hat{\imath}) + \tfrac{1}{2}\,(\hat{k}\times\hat{\jmath})\Bigr]
     = \frac{\mu_0 I_1 I_2 L}{2\pi d}
       \Bigl[-\tfrac{\sqrt{3}}{2}\,\hat{\jmath} - \tfrac{1}{2}\,\hat{\imath}\Bigr]
\end{align*}
\[
    \boxed{\vec{F}_{2\to1} = \frac{\mu_0 I_1 I_2 L}{2\pi d}
    \Bigl(-\tfrac{1}{2}\,\hat{\imath} - \tfrac{\sqrt{3}}{2}\,\hat{\jmath}\Bigr)}
\]
\begin{align*}
    \vec{F}_{3\to1} &= I_1 L\,\hat{k}\times\frac{\mu_0 I_3}{2\pi d}\,\hat{\jmath}
    = \frac{\mu_0 I_1 I_3 L}{2\pi d}\,(\hat{k}\times\hat{\jmath})
\end{align*}
\[
    \boxed{\vec{F}_{3\to1} = -\,\frac{\mu_0 I_1 I_3 L}{2\pi d}\,\hat{\imath}}
\]
\noindent\textbf{Interpretaci\'on:} (1) es antiparalela a (2) y a (3), por lo que ambas la
\textbf{repelen}: $\vec{F}_{2\to1}$ apunta desde (2) hacia (1) (a
$\theta + 180^\circ = 240^\circ$ del eje $+x$) y $\vec{F}_{3\to1}$ apunta en $-\hat{\imath}$.

%--------------------------------------------------
\subsection*{(c) Fuerza neta sobre la corriente (1) \ptos{6}}

\noindent Con $I_1 = 1$ [A], $I_2 = I_3 = 2$ [A], $L = 0{,}10$ [m] y $d = 0{,}20$ [m], el coeficiente
com\'un es:
\[
    \frac{\mu_0 I_1 I_2 L}{2\pi d} = \frac{\mu_0 I_1 I_3 L}{2\pi d}
    = \frac{\mu_0\,(1)(2)(0{,}10)}{2\pi\,(0{,}20)} = \frac{\mu_0}{2\pi}
\]
Por superposici\'on:
\begin{align*}
    \vec{F}_1 = \vec{F}_{2\to1} + \vec{F}_{3\to1}
    &= \frac{\mu_0}{2\pi}\Bigl(-\tfrac{1}{2}\,\hat{\imath} - \tfrac{\sqrt{3}}{2}\,\hat{\jmath}\Bigr)
       + \frac{\mu_0}{2\pi}\,(-\hat{\imath})
     = \frac{\mu_0}{2\pi}\Bigl(-\tfrac{3}{2}\,\hat{\imath} - \tfrac{\sqrt{3}}{2}\,\hat{\jmath}\Bigr)
\end{align*}
\[
    \boxed{\vec{F}_1 = \frac{\mu_0}{4\pi}\bigl(-3\,\hat{\imath} - \sqrt{3}\,\hat{\jmath}\bigr)\ [\text{N}]}
\]
\noindent Magnitud: $|\vec{F}_1| = \dfrac{\mu_0}{4\pi}\sqrt{9+3} = \dfrac{\mu_0\sqrt{12}}{4\pi}
= \dfrac{\sqrt{3}\,\mu_0}{2\pi}$ [N]. Num\'ericamente, con $\mu_0 = 4\pi\times10^{-7}$
[T$\cdot$m/A], $|\vec{F}_1| = 2\sqrt{3}\times10^{-7} = 3{,}46\times10^{-7}$ [N].

\section*{Problema 3: Fem inducida en una espira \textcolor{red}{(15 ptos)}}

\noindent Una espira plana de alambre de $8$ [cm$^2$] de \'area y $n = 6$ vueltas es perpendicular a
un campo magn\'etico que aumenta uniformemente de $0{,}5$ [T] a $2{,}5$ [T] en $1$ [s].
(a) Calcule la fem inducida en la espira. (b) Calcule la corriente inducida si la espira tiene una
resistencia de $4$ [$\Omega$].

\medskip

\noindent
\begin{minipage}[c]{0.36\textwidth}
\centering
\begin{tikzpicture}[scale=0.95, >=Stealth, line width=0.9pt]
    \foreach \x in {-1.2,-0.4,0.4,1.2}{
        \foreach \y in {-1.2,-0.4,0.4,1.2}{\punto[blue!70!black]{\x,\y}}}
    \draw[orange!80!black,line width=2pt,rounded corners=2pt] (-0.85,-0.85) rectangle (0.85,0.85);
    \node[orange!80!black,font=\small] at (0,-1.75){espira, $n = 6$};
    \node[blue!70!black,font=\small] at (1.75,1.55){$\vec{B}(t)$};
\end{tikzpicture}\\[2pt]
{\footnotesize Figura 3: $\vec{B}$ perpendicular al plano de la espira}
\end{minipage}%
\hfill
\begin{minipage}[c]{0.60\textwidth}
\textbf{Datos y transformaci\'on de unidades:}
\[
    A = 8\ [\text{cm}^2]\cdot\frac{10^{-4}\ [\text{m}^2]}{1\ [\text{cm}^2]} = 8\times10^{-4}\ [\text{m}^2]
\]
\[
    n = 6,\qquad B_i = 0{,}5\ [\text{T}],\qquad B_f = 2{,}5\ [\text{T}]
\]
\[
    \Delta t = 1\ [\text{s}],\qquad R = 4\ [\Omega]
\]
Como $\vec{B}$ es perpendicular al plano de la espira, es paralelo a su vector normal
($\theta = 0$, $\cos\theta = 1$).
\end{minipage}

%--------------------------------------------------
\subsection*{(a) Fem inducida \ptos{10}}

\noindent\textbf{Flujo a trav\'es de una vuelta:}
\[
    \Phi_B = \int_S\vec{B}\cdot d\vec{S} = B\,A\cos0^\circ = B\,A
\]
Como $A$ es constante y $B$ crece uniformemente, la tasa de cambio del campo es constante:
\[
    \frac{dB}{dt} = \frac{\Delta B}{\Delta t} = \frac{2{,}5 - 0{,}5}{1} = 2{,}0\ [\text{T/s}]
\]
\textbf{Ley de Faraday} para $n$ vueltas:
\begin{align*}
    \eps &= -\,n\,\frac{d\Phi_B}{dt} = -\,n\,A\,\frac{dB}{dt}
          = -(6)(8\times10^{-4})(2{,}0)
\end{align*}
\[
    \boxed{\eps = -9{,}6\times10^{-3}\ [\text{V}],\qquad |\eps| = 9{,}6\ [\text{mV}]}
\]
\noindent El signo negativo expresa la ley de Lenz: la fem inducida produce una corriente cuyo campo
se opone al \textbf{aumento} del flujo, es decir, un campo inducido opuesto a $\vec{B}$ en el interior
de la espira (en la Fig.\ 3, la corriente inducida circula en sentido \textbf{horario}).

%--------------------------------------------------
\subsection*{(b) Corriente inducida \ptos{5}}

\noindent Por la ley de Ohm:
\[
    i = \frac{|\eps|}{R} = \frac{9{,}6\times10^{-3}}{4}
    \qquad\Longrightarrow\qquad
    \boxed{i = 2{,}4\times10^{-3}\ [\text{A}] = 2{,}4\ [\text{mA}]}
\]
