%========================================================
% 1. MODULO DE dB
%========================================================
\noindent\textbf{1. M\'odulo del diferencial de campo.} Partimos de la ley de Biot--Savart en forma
vectorial,
\[
    d\vec{B} = \frac{\mu_0 I}{4\pi}\,
    \frac{d\vec{\ell}\times(\vec{r}-\vec{r}\,')}{\lVert\vec{r}-\vec{r}\,'\rVert^{3}},
\]
donde $d\vec{\ell}$ es el elemento diferencial de longitud del conductor, orientado en el sentido de
la corriente.

\noindent\textbf{Paso 1 (m\'odulo).} Con $\phi$ el \'angulo entre $d\vec{\ell}$ y $(\vec{r}-\vec{r}\,')$:
\[
    dB = \frac{\mu_0 I}{4\pi}\,
    \frac{d\ell\,\lVert\vec{r}-\vec{r}\,'\rVert\sin\phi}{\lVert\vec{r}-\vec{r}\,'\rVert^{3}}
    = \frac{\mu_0 I}{4\pi}\,\frac{d\ell\,\sin\phi}{\lVert\vec{r}-\vec{r}\,'\rVert^{2}}.
\]
\textbf{Paso 2 (geometr\'ia).} El campo se eval\'ua en el centro, $\vec{r}=\vec{0}$, de modo que
$\vec{r}-\vec{r}\,' = -\vec{r}\,'$ es siempre un vector \emph{radial}. Para un punto sobre un arco de
radio $R$, $\lVert\vec{r}-\vec{r}\,'\rVert = R$.

\noindent\textbf{Paso 3 (\'angulo entre $d\vec{\ell}$ y el vector radial).} Cada lazo tiene dos tipos
de tramo:

\noindent\textbf{3a) Arcos.} $d\vec{\ell}$ es \emph{tangente} al arco y $(\vec{r}-\vec{r}\,')$ es
\emph{radial}; son perpendiculares: $\phi=90^\circ$, $\sin\phi=1$.

\noindent\textbf{3b) Tramos rectos radiales.} En los dos segmentos que unen el arco interior con el
exterior, $d\vec{\ell}$ apunta \emph{a lo largo del radio}: hacia afuera o hacia adentro, seg\'un el
sentido de la corriente. Como $(\vec{r}-\vec{r}\,')=-\vec{r}\,'$ tambi\'en es radial, ambos vectores
son paralelos o antiparalelos: $\phi = 0^\circ$ o $\phi = 180^\circ$. En los dos casos:
\[
    \lVert d\vec{\ell}\times(\vec{r}-\vec{r}\,')\rVert
    = d\ell\,\lVert\vec{r}-\vec{r}\,'\rVert\,\sin\phi = 0
    \quad\Longrightarrow\quad d\vec{B} = \vec{0}.
\]
Los tramos radiales \textbf{no aportan} campo en $O$; solo contribuyen los arcos.

\noindent\textbf{Paso 4 (elemento de arco).} En cada arco, el elemento $d\ell$ es el
\textbf{diferencial de longitud de arco de circunferencia}: un arco de radio $R$ que subtiende el
\'angulo $d\theta$ mide
\[
    d\ell = R\,d\theta \qquad(\text{arco} = \text{radio}\times\text{\'angulo}).
\]

\noindent\textbf{Paso 5 (sustituci\'on e integraci\'on).}
\[
    dB = \frac{\mu_0 I}{4\pi}\,\frac{R\,d\theta}{R^{2}} = \frac{\mu_0 I}{4\pi R}\,d\theta
    \quad\Longrightarrow\quad
    \boxed{B_{\text{arco}} = \frac{\mu_0 I}{4\pi R}\int_{\text{arco}} d\theta = \frac{\mu_0 I\,\Delta\theta}{4\pi R}}
\]
\noindent\textit{Observaci\'on clave:} el m\'odulo depende \textbf{solo} del radio $R$ y de la
apertura $\Delta\theta$, no de d\'onde comienza o termina el arco.

\noindent\textbf{Direcci\'on (regla de la mano derecha).} Se curvan los dedos de la mano derecha en
el sentido en que circula la corriente por el arco; el \textbf{pulgar} apunta en la direcci\'on del
campo magn\'etico en el centro $O$. Para un arco, el campo en $O$ siempre es perpendicular al plano
del arco.

\secgap

%========================================================
% 2. LAZO 1
%========================================================
\noindent\textbf{2. (a) Lazo 1 (plano $xy$, $45^\circ$).}

\noindent\textbf{M\'odulos de sus arcos} ($\Delta\theta_1 = \pi/4$):
\begin{align*}
    B_{R_1} &= \frac{\mu_0 I}{4\pi R_1}\cdot\frac{\pi}{4} = \frac{\mu_0 I}{16\,R_1}
             = \frac{(4\pi\times10^{-7})(1{,}5)}{16\,(0{,}03)} = 3{,}927\times10^{-6}\ [\text{T}],\\
    B_{R_2} &= \frac{\mu_0 I}{16\,R_2}
             = \frac{(4\pi\times10^{-7})(1{,}5)}{16\,(0{,}06)} = 1{,}963\times10^{-6}\ [\text{T}].
\end{align*}

\noindent\textbf{Direcciones (regla de la mano derecha, mirando desde $+z$):}
\begin{itemize}\setlength\itemsep{2pt}
    \item Arco exterior ($R_2$), \textbf{antihorario}: los dedos giran en sentido antihorario y el
          pulgar apunta hacia el observador, es decir, hacia $+z$ $\Rightarrow$ $+\kk$.
    \item Arco interior ($R_1$), \textbf{horario}: los dedos giran en sentido horario y el pulgar se
          aleja del observador, hacia $-z$ $\Rightarrow$ $-\kk$.
\end{itemize}

\noindent\textbf{Campo del lazo 1:}
\[
    \vec{B}_1 = (B_{R_2}-B_{R_1})\,\kk
    = \frac{\mu_0 I}{16}\left(\frac{1}{R_2}-\frac{1}{R_1}\right)\kk
    = -\frac{\mu_0 I\,(R_2-R_1)}{16\,R_1R_2}\,\kk .
\]
Con $R_2 = 2R_1$ se simplifica a $\vec{B}_1 = -\dfrac{\mu_0 I}{32\,R_1}\,\kk$; num\'ericamente,
$\vec{B}_1 = (1{,}963-3{,}927)\times10^{-6}\,\kk$:
\[
    \boxed{\vec{B}_1 = -1{,}963\times10^{-6}\,\kk\ [\text{T}]}
\]
Como $R_1<R_2$, domina el arco interior y $\vec{B}_1$ apunta a $-\kk$.

\secgap

%========================================================
% 3. LAZO 2
%========================================================
\noindent\textbf{3. (b) Lazo 2 (plano $zx$, $60^\circ$).}

\noindent\textbf{M\'odulos de sus arcos} ($\Delta\theta_2 = \pi/3$):
\begin{align*}
    B_{R_3} &= \frac{\mu_0 I}{4\pi R_3}\cdot\frac{\pi}{3} = \frac{\mu_0 I}{12\,R_3}
             = \frac{(4\pi\times10^{-7})(1{,}5)}{12\,(0{,}07)} = 2{,}244\times10^{-6}\ [\text{T}],\\
    B_{R_4} &= \frac{\mu_0 I}{12\,R_4}
             = \frac{(4\pi\times10^{-7})(1{,}5)}{12\,(0{,}21)} = 0{,}748\times10^{-6}\ [\text{T}].
\end{align*}

\noindent\textbf{Direcciones (regla de la mano derecha, mirando desde $+y$):}
\begin{itemize}\setlength\itemsep{2pt}
    \item Arco exterior ($R_4$), \textbf{horario}: los dedos giran en sentido horario y el pulgar se
          aleja del observador, hacia $-y$ $\Rightarrow$ $-\jj$.
    \item Arco interior ($R_3$), \textbf{antihorario}: los dedos giran en sentido antihorario y el
          pulgar apunta hacia el observador, hacia $+y$ $\Rightarrow$ $+\jj$.
\end{itemize}

\noindent\textbf{Campo del lazo 2:}
\[
    \vec{B}_2 = (B_{R_3}-B_{R_4})\,\jj
    = \frac{\mu_0 I}{12}\left(\frac{1}{R_3}-\frac{1}{R_4}\right)\jj
    = \frac{\mu_0 I\,(R_4-R_3)}{12\,R_3R_4}\,\jj .
\]
Con $R_4 = 3R_3$ se simplifica a $\vec{B}_2 = \dfrac{\mu_0 I}{18\,R_3}\,\jj$; num\'ericamente,
$\vec{B}_2 = (2{,}244-0{,}748)\times10^{-6}\,\jj$:
\[
    \boxed{\vec{B}_2 = 1{,}496\times10^{-6}\,\jj\ [\text{T}]}
\]
Como $R_3<R_4$, domina el arco interior y $\vec{B}_2$ apunta a $+\jj$.

%========================================================
% 5. INTERPRETACION
%========================================================
\noindent\textbf{5. Interpretaci\'on f\'isica.}
\begin{itemize}\itemsep6pt
    \item \textbf{Cancelaci\'on parcial dentro de cada lazo.} En un lazo cerrado el arco interior y
    el exterior recorren sentidos opuestos, por lo que sus campos en $O$ se \emph{restan}. Domina el
    arco m\'as cercano a $O$ (radio menor). Si los dos radios de un lazo fueran iguales, su campo
    ser\'ia nulo.
    \item \textbf{Los tramos radiales no cuentan.} Su $d\vec{\ell}$ es paralelo o antiparalelo al
    vector radial ($\phi = 0^\circ$ o $180^\circ$), y el producto cruz se anula
    (paso~3b).
    \item \textbf{Planos perpendiculares, campos perpendiculares.} El lazo 1 ($xy$) produce un
    campo a lo largo de $\pm\kk$ y el lazo 2 ($zx$) a lo largo de $\pm\jj$. Por eso se suman
    \emph{vectorialmente} y $\vec{B}$ queda en el plano $yz$. Si ambos lazos estuvieran en el
    \emph{mismo} plano, como en la Tarea~4, sus campos ser\'ian colineales y se sumar\'ian como
    escalares.
    \item \textbf{Posici\'on de los arcos.} Girar un lazo dentro de su plano, manteniendo radios,
    apertura y sentido de circulaci\'on, no cambia el campo en $O$. Solo importan los radios,
    $\Delta\theta$, el plano y el sentido de la corriente.
\end{itemize}
