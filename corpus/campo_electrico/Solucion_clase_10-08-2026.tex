%========================================================
% ENUNCIADO
%========================================================
\noindent\textbf{Enunciado.} Cuatro cargas id\'enticas $q = 15\ [\text{mC}]$ se ubican en los v\'ertices
de un rect\'angulo de lados $L = 20\ [\text{cm}]$ y $W = 65\ [\text{cm}]$. Determine la fuerza el\'ectrica
resultante sobre la carga de la \textbf{esquina superior derecha}.

\medskip

\noindent

\begin{minipage}[t]{0.52\textwidth}
\vspace{0pt}
\textbf{Numeraci\'on y estrategia.} Las cargas se numeran del 1 al 4 en sentido
\textbf{antihorario}, partiendo por la carga del origen como $q_1$. As\'i el objetivo (esquina
superior derecha) es $q_3$.

\smallskip
Todas las cargas son \emph{id\'enticas y positivas}, luego las tres fuerzas sobre $q_3$ son
\textbf{repulsivas}. Se usa la forma vectorial de Coulomb:
\[
    \vec F_{3j} = k\,q^2\,\frac{\vec r_3-\vec r_j}{|\vec r_3-\vec r_j|^3},
\]
con $k\,q^2 = 9\times10^{9}(0{,}015)^2 = 2{,}025\times10^{6}\ [\text{N m}^2]$.
\end{minipage}

\medskip

\noindent\textbf{Coordenadas de cada carga} (en metros, en funci\'on de los unitarios cartesianos):
\begin{align*}
    \vec r_1 &= \vec 0 & &(\text{esquina inf.\ izquierda, origen}),\\
    \vec r_2 &= L\,\hat\imath = 0{,}20\,\hat\imath & &(\text{esquina inf.\ derecha}),\\
    \vec r_3 &= L\,\hat\imath + W\,\hat\jmath = 0{,}20\,\hat\imath + 0{,}65\,\hat\jmath & &(\text{esquina sup.\ derecha, objetivo}),\\
    \vec r_4 &= W\,\hat\jmath = 0{,}65\,\hat\jmath & &(\text{esquina sup.\ izquierda}).
\end{align*}

\noindent\textbf{Restas de vectores} (posici\'on de $q_3$ respecto a cada fuente):
\begin{align*}
    \vec r_3-\vec r_1 &= (0{,}20\,\hat\imath + 0{,}65\,\hat\jmath) - \vec 0 = 0{,}20\,\hat\imath + 0{,}65\,\hat\jmath,
        & |\vec r_3-\vec r_1| &= \sqrt{0{,}20^2+0{,}65^2} = 0{,}680\ \text{m},\\[2pt]
    \vec r_3-\vec r_2 &= (0{,}20\,\hat\imath + 0{,}65\,\hat\jmath) - 0{,}20\,\hat\imath = 0{,}65\,\hat\jmath,
        & |\vec r_3-\vec r_2| &= 0{,}65\ \text{m},\\[2pt]
    \vec r_3-\vec r_4 &= (0{,}20\,\hat\imath + 0{,}65\,\hat\jmath) - 0{,}65\,\hat\jmath = 0{,}20\,\hat\imath,
        & |\vec r_3-\vec r_4| &= 0{,}20\ \text{m}.
\end{align*}

\noindent\textbf{Fuerza de $q_4$} (esquina sup.\ izquierda; $\vec r_3-\vec r_4$ horizontal):
\[
    \vec F_{34} = k\,q^2\,\frac{0{,}20\,\hat\imath}{(0{,}20)^3}
    = \frac{2{,}025\times10^{6}}{(0{,}20)^2}\,\hat\imath = 5{,}06\times10^{7}\,\hat\imath\ [\text{N}]
\]

\noindent\textbf{Fuerza de $q_2$} (esquina inf.\ derecha; $\vec r_3-\vec r_2$ vertical):
\[
    \vec F_{32} = k\,q^2\,\frac{0{,}65\,\hat\jmath}{(0{,}65)^3}
    = \frac{2{,}025\times10^{6}}{(0{,}65)^2}\,\hat\jmath = 4{,}79\times10^{6}\,\hat\jmath\ [\text{N}]
\]

\noindent\textbf{Fuerza de $q_1$} (esquina inf.\ izquierda; $\vec r_3-\vec r_1$ diagonal):
\begin{align*}
    \vec F_{31} &= k\,q^2\,\frac{0{,}20\,\hat\imath + 0{,}65\,\hat\jmath}{(0{,}680)^3}
    = \frac{2{,}025\times10^{6}}{(0{,}680)^3}\,(0{,}20\,\hat\imath + 0{,}65\,\hat\jmath)\\[2pt]
    &= 1{,}29\times10^{6}\,\hat\imath + 4{,}18\times10^{6}\,\hat\jmath\ [\text{N}]
    \qquad(|\vec F_{31}| = 4{,}38\times10^{6}\ [\text{N}])
\end{align*}

\noindent\textbf{Resultante} (superposici\'on $\vec F_3 = \vec F_{31}+\vec F_{32}+\vec F_{34}$):
\begin{align*}
    F_x &= 5{,}06\times10^{7} + 1{,}29\times10^{6} = 5{,}19\times10^{7}\ [\text{N}],\\[2pt]
    F_y &= 4{,}79\times10^{6} + 4{,}18\times10^{6} = 8{,}98\times10^{6}\ [\text{N}].
\end{align*}
\[
    \boxed{\vec F_3 = 5{,}19\times10^{7}\,\hat\imath + 8{,}98\times10^{6}\,\hat\jmath\ [\text{N}],
    \qquad |\vec F_3| = 5{,}27\times10^{7}\ [\text{N}]}
\]

\noindent La resultante forma un \'angulo
$\theta = \arctan\!\big(\tfrac{8{,}98\times10^{6}}{5{,}19\times10^{7}}\big) = 9{,}8^\circ$ sobre el eje $x$,
dominada por la carga vecina $q_4$, que es la m\'as cercana ($L\ll W$).
