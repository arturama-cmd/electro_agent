\section*{Problema 3 (Unidimensional)}

\subsection*{Enunciado}

Dado el sistema de la Fig.~1, determine la fuerza el\'ectrica que siente una carga $q = 5$~[nC], si \'esta se ubica en el punto $A$ y luego en el punto $B$. Datos:
\[
q_1 = -12{,}5\ \text{[nC]}, \quad q_2 = -6{,}25\ \text{[nC]}, \quad d_{12} = 0{,}25\ \text{[m]}
\]

\subsection*{Diagrama}

\begin{center}
\begin{tikzpicture}[scale=1.1, >=Stealth]
    \draw[->, gray, thin] (-1.8,0) -- (3.2,0) node[right]{\small $x$};
    \fill[cargarojo] (0,0) circle (5pt) node[above=5pt]{\small $q_2{=}{-}6{,}25\ \text{nC}$};
    \fill[cargarojo] (2.5,0) circle (5pt) node[above=5pt]{\small $q_1{=}{-}12{,}5\ \text{nC}$};
    \fill[cargaverde] (1.5,0) circle (4pt) node[below=5pt]{\small $A$};
    \fill[cargaverde] (-1.0,0) circle (4pt) node[below=5pt]{\small $B$};
    \draw[<->, black, thin] (-1.0,-0.55) -- (0,-0.55) node[midway, below]{\small $0{,}10$~m};
    \draw[<->, black, thin] (1.5,-0.55) -- (2.5,-0.55) node[midway, below]{\small $0{,}10$~m};
    \draw[<->, black, thin] (0,-0.90) -- (2.5,-0.90) node[midway, below]{\small $0{,}25$~m};
    \node[font=\scriptsize,gray] at (0,0.62){origen};
\end{tikzpicture}
\end{center}

\subsection*{Vectores de posici\'on}

Con el origen en $q_2$ (todo sobre el eje $x$):
\[
\vec r_{q_2} = \vec 0,\qquad \vec r_{q_1} = 0{,}25\,\hat\imath,\qquad
\vec r_A = 0{,}15\,\hat\imath,\qquad \vec r_B = -0{,}10\,\hat\imath \quad[\text{m}]
\]
La fuerza sobre $q$ es $\vec F = \vec F_{q_1} + \vec F_{q_2}$, con
$\vec F_{q_j}=k\,\dfrac{q\,q_j}{\lVert\vec r-\vec r_{q_j}\rVert^{3}}(\vec r-\vec r_{q_j})$,
$\ k\,q\,q_1=-5{,}625\times10^{-7}$, $\ k\,q\,q_2=-2{,}8125\times10^{-7}$.

\subsection*{Punto $A$} ($\vec r_A=0{,}15\,\hat\imath$):
\begin{align*}
\vec r_A-\vec r_{q_1} &= 0{,}15\,\hat\imath-0{,}25\,\hat\imath = -0{,}10\,\hat\imath, & \lVert\cdot\rVert &= 0{,}10,\\
\vec r_A-\vec r_{q_2} &= 0{,}15\,\hat\imath-\vec 0 = 0{,}15\,\hat\imath, & \lVert\cdot\rVert &= 0{,}15.
\end{align*}
\begin{align*}
\vec F_{q_1} &= \frac{-5{,}625\times10^{-7}}{(0{,}10)^3}\,(-0{,}10\,\hat\imath) = 5{,}625\times10^{-5}\,\hat\imath\ [\text{N}],\\[2pt]
\vec F_{q_2} &= \frac{-2{,}8125\times10^{-7}}{(0{,}15)^3}\,(0{,}15\,\hat\imath) = -1{,}25\times10^{-5}\,\hat\imath\ [\text{N}].
\end{align*}
\[
\boxed{\vec F^{(A)} = (5{,}625-1{,}25)\times10^{-5}\,\hat\imath = 4{,}375\times10^{-5}\,\hat\imath\ [\text{N}]}
\]

\subsection*{Punto $B$} ($\vec r_B=-0{,}10\,\hat\imath$):
\begin{align*}
\vec r_B-\vec r_{q_1} &= -0{,}10\,\hat\imath-0{,}25\,\hat\imath = -0{,}35\,\hat\imath, & \lVert\cdot\rVert &= 0{,}35,\\
\vec r_B-\vec r_{q_2} &= -0{,}10\,\hat\imath-\vec 0 = -0{,}10\,\hat\imath, & \lVert\cdot\rVert &= 0{,}10.
\end{align*}
\begin{align*}
\vec F_{q_1} &= \frac{-5{,}625\times10^{-7}}{(0{,}35)^3}\,(-0{,}35\,\hat\imath) = 4{,}592\times10^{-6}\,\hat\imath\ [\text{N}],\\[2pt]
\vec F_{q_2} &= \frac{-2{,}8125\times10^{-7}}{(0{,}10)^3}\,(-0{,}10\,\hat\imath) = 2{,}8125\times10^{-5}\,\hat\imath\ [\text{N}].
\end{align*}
\[
\boxed{\vec F^{(B)} = (0{,}4592+2{,}8125)\times10^{-5}\,\hat\imath = 3{,}272\times10^{-5}\,\hat\imath\ [\text{N}]}
\]

\noindent\textbf{Interpretaci\'on.} Ambas fuentes son negativas y $q$ positiva: atraen a $q$. En $A$ (entre las cargas) las atracciones se oponen y domina $q_1$ (m\'as cercana e intensa) $\to+\hat\imath$. En $B$ (a la izquierda de ambas) las dos atraen hacia $+\hat\imath$ y se suman.

\section*{Problema 1 (Bidimensional)}

\subsection*{Enunciado}

En los v\'ertices de un cuadrado de lado $a = 5{,}5$~[cm] se ubican cuatro cargas puntuales id\'enticas $q = -5\ [\mu\text{C}]$. Determine la fuerza el\'ectrica sobre la carga de la esquina inferior izquierda.

\medskip

\noindent
\begin{minipage}[t]{0.42\textwidth}
\vspace{0pt}
\centering
\begin{tikzpicture}[scale=2.4, >=Stealth]
    \coordinate (BL) at (0,0); \coordinate (BR) at (1,0);
    \coordinate (TL) at (0,1); \coordinate (TR) at (1,1);
    \draw[gray, dashed, thin] (BL) -- (BR) -- (TR) -- (TL) -- cycle;
    \draw[gray, dashed, very thin] (BL) -- (TR);
    \draw[<->, black, thin] (0,-0.16) -- (1,-0.16) node[midway, below]{\small $a$};
    \draw[<->, black, thin] (1.16,0) -- (1.16,1) node[midway, right]{\small $a$};
    % numeracion antihoraria desde el origen (q1 = objetivo, inf. izq.)
    \foreach \pos/\lbl in {(BR)/$q_2$, (TR)/$q_3$, (TL)/$q_4$} {
        \fill[cargarojo] \pos circle (2.3pt);
        \node[cargarojo, font=\small] at \pos [above right=1pt] {\lbl};
    }
    \fill[cargazul] (BL) circle (3pt);
    \node[cargazul, font=\small, above right=1pt] at (BL) {$q_1$};
    % fuerzas sobre q1
    \draw[->, cargarojo, very thick] (0,0) -- (-0.40,0) node[left]{\small $\vec{F}_{12}$};
    \draw[->, cargarojo, very thick] (0,0) -- (0,-0.40) node[below]{\small $\vec{F}_{14}$};
    \draw[->, cargarojo, very thick] (0,0) -- (-0.27,-0.27) node[below left]{\small $\vec{F}_{13}$};
    % ejes
    \draw[->,blue,thin] (0.62,0.30)--(0.85,0.30) node[right,font=\scriptsize]{$\hat\imath$};
    \draw[->,blue,thin] (0.62,0.30)--(0.62,0.53) node[above,font=\scriptsize]{$\hat\jmath$};
\end{tikzpicture}\\[2pt]
{\scriptsize Numeraci\'on antihoraria desde el origen ($q_1$ = objetivo).}
\end{minipage}%
\hfill
\begin{minipage}[t]{0.54\textwidth}
\vspace{0pt}
\textbf{Vectores de posici\'on} (origen en $q_1$, objetivo; numeraci\'on antihoraria), $a=0{,}055$ m:
\[
\vec r_1=\vec 0,\ \ \vec r_2=a\,\hat\imath,\ \ \vec r_3=a\,\hat\imath+a\,\hat\jmath,\ \ \vec r_4=a\,\hat\jmath
\]
Las cuatro cargas son iguales ($q$), luego $q_1 q_j=q^2>0$: $q_1$ es \textbf{repelida} por las otras tres.
\[
\vec F_1 = k\,q^2\!\left[\frac{\vec r_1-\vec r_2}{\lVert\cdot\rVert^3}+\frac{\vec r_1-\vec r_3}{\lVert\cdot\rVert^3}+\frac{\vec r_1-\vec r_4}{\lVert\cdot\rVert^3}\right]
\]
\end{minipage}

\medskip
\noindent\textbf{Restas de vectores:}
\[
\vec r_1-\vec r_2 = -a\,\hat\imath\ (a),\qquad
\vec r_1-\vec r_3 = -a\,\hat\imath-a\,\hat\jmath\ (a\sqrt2),\qquad
\vec r_1-\vec r_4 = -a\,\hat\jmath\ (a)
\]
\begin{align*}
\vec F_{12} &= \frac{k q^2}{a^3}(-a\,\hat\imath) = \frac{k q^2}{a^2}(-\hat\imath),\qquad
\vec F_{14} = \frac{k q^2}{a^2}(-\hat\jmath),\\[2pt]
\vec F_{13} &= \frac{k q^2}{(a\sqrt2)^3}(-a\,\hat\imath-a\,\hat\jmath)
           = \frac{k q^2}{a^2\,2^{3/2}}(-\hat\imath-\hat\jmath).
\end{align*}
Sumando, con $1/2^{3/2}=0{,}3536$ y $\dfrac{k q^2}{a^2}=9\times10^9\dfrac{(5\times10^{-6})^2}{(0{,}055)^2}=74{,}38$~N:
\[
\vec F_1 = \frac{k q^2}{a^2}\Big[-\big(1+\tfrac{1}{2^{3/2}}\big)\hat\imath-\big(1+\tfrac{1}{2^{3/2}}\big)\hat\jmath\Big]
= 74{,}38\,(-1{,}3536)(\hat\imath+\hat\jmath)
\]
\[
\boxed{\vec F_1 = -100{,}68\,\hat\imath - 100{,}68\,\hat\jmath\ [\text{N}],\qquad |\vec F_1| = 142{,}4\ [\text{N}]}
\]
\noindent\textbf{Interpretaci\'on.} Por simetr\'ia la resultante va por la diagonal hacia el exterior del cuadrado ($225^\circ$). La carga diagonal $q_3$ aporta menos (distancia $a\sqrt2$) pero refuerza la direcci\'on.

\section*{Problema 2 (Bidimensional)}

\subsection*{Enunciado}

Dipolo el\'ectrico: $q_1=+5$~nC en el origen y $q_2=-5$~nC en $(0{,}10,\,0)$~m. Determine la fuerza sobre $q_0=3$~[nC] en los puntos $a$, $b$ y $c$.

\subsection*{Diagrama}

\begin{center}
\begin{tikzpicture}[scale=2.0, >=Stealth]
    \fill[cargazul]  (0,0) circle (2.5pt) node[below=5pt]{\small $q_1{=}{+}5$~nC};
    \fill[cargarojo] (1,0) circle (2.5pt) node[below=5pt]{\small $q_2{=}{-}5$~nC};
    \draw[gray, dashed, thin] (-0.7,0) -- (1.5,0) node[right]{\small $x$};
    \fill[cargaverde] (0.6,0) circle (2pt) node[above=4pt]{\small $a$};
    \draw[<->, black!50, thin] (0,0.28) -- (0.6,0.28) node[midway,above]{\scriptsize $0{,}06$~m};
    \draw[<->, black!50, thin] (0.6,0.28) -- (1.0,0.28) node[midway,above]{\scriptsize $0{,}04$~m};
    \fill[cargaverde] (-0.4,0) circle (2pt) node[above=4pt]{\small $b$};
    \draw[<->, black!50, thin] (-0.4,-0.44) -- (0,-0.44) node[midway,below]{\scriptsize $0{,}04$~m};
    \fill[cargaverde] (0.5, 0.96) circle (2pt) node[right=3pt]{\small $c$};
    \draw[black!40, thin] (0,0) -- (0.5,0.96);
    \draw[black!40, thin] (1,0) -- (0.5,0.96);
    \node[black!60, font=\scriptsize] at (0.07,0.55) {$0{,}13$~m};
    \node[black!60, font=\scriptsize] at (0.93,0.55) {$0{,}13$~m};
    \node[font=\scriptsize] at (0.5, -0.62) {$\underbrace{\hspace{2.2cm}}_{d=0{,}10\ \text{m}}$};
\end{tikzpicture}
\end{center}

\noindent\textbf{Vectores de posici\'on:} $\vec r_1=\vec 0$, $\vec r_2=0{,}10\,\hat\imath$;
$\vec r_a=0{,}06\,\hat\imath$, $\vec r_b=-0{,}04\,\hat\imath$, $\vec r_c=0{,}05\,\hat\imath+0{,}12\,\hat\jmath$ [m].
La fuerza sobre $q_0$ es $\vec F=\vec F_1+\vec F_2$ con
$\vec F_j=k\,\dfrac{q_0 q_j}{\lVert\vec r-\vec r_j\rVert^3}(\vec r-\vec r_j)$,
$\ k q_0 q_1=1{,}35\times10^{-7}$, $\ k q_0 q_2=-1{,}35\times10^{-7}$.

\subsection*{Punto $a$} ($\vec r_a-\vec r_1=0{,}06\,\hat\imath$; $\ \vec r_a-\vec r_2=-0{,}04\,\hat\imath$):
\[
\vec F_a = \frac{1{,}35\times10^{-7}}{(0{,}06)^3}(0{,}06\,\hat\imath)+\frac{-1{,}35\times10^{-7}}{(0{,}04)^3}(-0{,}04\,\hat\imath)
= (3{,}75+8{,}44)\times10^{-5}\,\hat\imath
\]
\[
\boxed{\vec F_a = 1{,}219\times10^{-4}\,\hat\imath\ [\text{N}]}
\]

\subsection*{Punto $b$} ($\vec r_b-\vec r_1=-0{,}04\,\hat\imath$; $\ \vec r_b-\vec r_2=-0{,}14\,\hat\imath$):
\[
\vec F_b = \frac{1{,}35\times10^{-7}}{(0{,}04)^3}(-0{,}04\,\hat\imath)+\frac{-1{,}35\times10^{-7}}{(0{,}14)^3}(-0{,}14\,\hat\imath)
= (-8{,}438+0{,}689)\times10^{-5}\,\hat\imath
\]
\[
\boxed{\vec F_b = -7{,}749\times10^{-5}\,\hat\imath\ [\text{N}]}
\]

\subsection*{Punto $c$} ($\vec r_c-\vec r_1=0{,}05\,\hat\imath+0{,}12\,\hat\jmath$; $\ \vec r_c-\vec r_2=-0{,}05\,\hat\imath+0{,}12\,\hat\jmath$; ambos de norma $0{,}13$):
\begin{align*}
\vec F_1 &= \frac{1{,}35\times10^{-7}}{(0{,}13)^3}(0{,}05\,\hat\imath+0{,}12\,\hat\jmath) = 3{,}072\times10^{-6}\,\hat\imath+7{,}374\times10^{-6}\,\hat\jmath,\\[2pt]
\vec F_2 &= \frac{-1{,}35\times10^{-7}}{(0{,}13)^3}(-0{,}05\,\hat\imath+0{,}12\,\hat\jmath) = 3{,}072\times10^{-6}\,\hat\imath-7{,}374\times10^{-6}\,\hat\jmath.
\end{align*}
Las componentes $\hat\jmath$ se cancelan por simetr\'ia:
\[
\boxed{\vec F_c = 6{,}144\times10^{-6}\,\hat\imath\ [\text{N}]}
\]

\noindent\textbf{Interpretaci\'on.} En $a$ (entre las cargas) repulsi\'on de $q_1$ y atracci\'on de $q_2$ apuntan igual ($+\hat\imath$): fuerza m\'axima. En $b$ domina la repulsi\'on de $q_1$ ($-\hat\imath$). En $c$ (mediatriz) las componentes verticales se anulan y solo queda $+\hat\imath$ (la menor).

\section*{Problema 4 (Bidimensional)}

\subsection*{Enunciado}

Dos cargas $q_1$ y $q_2$ separadas $4{,}5$~[cm]. Una carga $Q = -1{,}75\ [\mu\text{C}]$, masa $m = 5$~[g], equidistante $3$~[cm] de cada una, se suelta desde el reposo y adquiere aceleraci\'on $a_y = 324\ [\text{m/s}^2]$ hacia arriba. Determine $q_1$ y $q_2$.

\medskip

\noindent
\begin{minipage}[t]{0.40\textwidth}
\vspace{0pt}
\centering
\begin{tikzpicture}[scale=0.95, >=Stealth]
    \coordinate (Qpos) at (0,0); \coordinate (q1pos) at (1.984, 2.25); \coordinate (q2pos) at (1.984,-2.25);
    \draw[black!40, thin] (Qpos) -- (q1pos); \draw[black!40, thin] (Qpos) -- (q2pos);
    \draw[dashed, black!25, thin] (Qpos) -- (1.984, 0);
    \draw[dashed, black!25, thin] (q1pos) -- (q2pos);
    \draw[gray!70, thin] (0.50, 0) arc (0:48.6:0.50);
    \node[font=\scriptsize, gray] at (0.66, 0.24) {$\alpha$};
    \draw[<->, thin, black!60] (2.65,-2.25) -- (2.65, 2.25) node[midway, right, font=\small]{$4{,}50$~cm};
    \node[above left, font=\scriptsize] at (0.99, 1.13) {$3{,}00$~cm};
    \node[below left, font=\scriptsize] at (0.99,-1.13) {$3{,}00$~cm};
    \fill[cargazul]  (Qpos)  circle (3pt) node[left=4pt, font=\small]{$Q(-)$};
    \fill[cargazul!60!white] (q1pos) circle (3pt) node[right=3pt, font=\small]{$q_1(+)$};
    \fill[cargarojo!60!white] (q2pos) circle (3pt) node[right=3pt, font=\small]{$q_2(-)$};
    \draw[->, cargazul, very thick] (Qpos) -- +(0.529, 0.600) node[right, font=\scriptsize]{$\vec{F}_1$};
    \draw[->, cargarojo, very thick] (Qpos) -- +(-0.529, 0.600) node[left, font=\scriptsize]{$\vec{F}_2$};
    \draw[->, cargaverde, very thick] (Qpos) -- +(0, 1.05) node[right, font=\scriptsize]{$\vec{F}_T{=}m\vec a$};
\end{tikzpicture}
\end{minipage}%
\hfill
\begin{minipage}[t]{0.56\textwidth}
\vspace{0pt}
\textbf{Geometr\'ia.} $Q$ en el origen; $d_x=\sqrt{3^2-2{,}25^2}=1{,}984$~cm. Vectores de posici\'on (m):
\[
\vec r_Q=\vec 0,\quad \vec r_1=0{,}01984\,\hat\imath+0{,}0225\,\hat\jmath,\quad \vec r_2=0{,}01984\,\hat\imath-0{,}0225\,\hat\jmath
\]
\textbf{Restas} (ambas de norma $0{,}03$~m):
\[
\vec r_Q-\vec r_1=-0{,}01984\,\hat\imath-0{,}0225\,\hat\jmath,\quad
\vec r_Q-\vec r_2=-0{,}01984\,\hat\imath+0{,}0225\,\hat\jmath
\]
\end{minipage}

\medskip
\noindent Fuerza sobre $Q$ ($\vec F_Q=\vec F_1+\vec F_2$), con
$\dfrac{k\,Q}{(0{,}03)^3}=\dfrac{9\times10^9(-1{,}75\times10^{-6})}{2{,}7\times10^{-5}}=-5{,}833\times10^{8}$:
\begin{align*}
\vec F_Q &= \frac{k\,Q}{(0{,}03)^3}\Big[q_1(-0{,}01984\,\hat\imath-0{,}0225\,\hat\jmath)+q_2(-0{,}01984\,\hat\imath+0{,}0225\,\hat\jmath)\Big]\\[2pt]
&= \underbrace{1{,}158\times10^{7}(q_1+q_2)}_{F_x}\,\hat\imath + \underbrace{1{,}3125\times10^{7}(q_1-q_2)}_{F_y}\,\hat\jmath.
\end{align*}
\textbf{Condici\'on.} La aceleraci\'on es solo vertical $\Rightarrow F_x=0 \Rightarrow q_1+q_2=0 \Rightarrow q_2=-q_1$
(cargas de igual magnitud y signo opuesto: $q_1$ positiva, $q_2$ negativa). Entonces
\[
F_y = 1{,}3125\times10^{7}\,(q_1-q_2) = 1{,}3125\times10^{7}\,(2q_1) = m\,a_y = (0{,}005)(324) = 1{,}62\ \text{N}
\]
\[
q_1 = \frac{1{,}62}{2{,}625\times10^{7}} = 6{,}17\times10^{-8}\ \text{C}
\]
\[
\boxed{q_1 = +61{,}7\ \text{[nC]},\qquad q_2 = -61{,}7\ \text{[nC]}}
\]
\noindent\textbf{Interpretaci\'on.} La simetr\'ia ($F_x=0$) obliga a $q_2=-q_1$: $q_1$ atrae a $Q$ y $q_2$ lo repele, con componentes horizontales opuestas que se cancelan y verticales (hacia arriba) que se suman, produciendo la aceleraci\'on observada.
