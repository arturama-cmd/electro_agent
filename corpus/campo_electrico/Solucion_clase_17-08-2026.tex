{\centering\Large\textbf{Problema Resuelto en la clase del 17/08}\par}
{\centering\normalsize\textit{F\'isica II -- Electromagnetismo \quad(Campo el\'ectrico de cargas puntuales)}\par}

\vspace{0.5cm}
{\centering\footnotesize (Constante de Coulomb $k=9\times10^{9}\ \mathrm{N\,m^2/C^2}$;
carga elemental $e=1{,}6\times10^{-19}$ C; $1\ \mu C=10^{-6}$ C)\par}

\vspace{0.5cm}

%========================================================
% ENUNCIADO
%========================================================
\noindent\textbf{Enunciado.} Cuatro cargas puntuales se ubican en los v\'ertices de un cuadrado de
lado $a = 4\ [\text{cm}]$: $2q$ (sup.\ izq.), $q$ (sup.\ der.), $3q$ (inf.\ izq.) y $4q$ (inf.\ der.),
con $q = 5\ [\mu C]$. Determine el \textbf{campo el\'ectrico $\vec E$ en el centro} del cuadrado y la
\textbf{fuerza} que sentir\'ia all\'i un \textbf{electr\'on} y un \textbf{prot\'on}.

\medskip

\noindent
\begin{minipage}[t]{0.40\textwidth}
\centering
\vspace{0pt}

\begin{minipage}[t]{0.56\textwidth}
\vspace{0pt}
\textbf{Vectores de posici\'on} (origen en $3q$; $a=0{,}04$ m):
\begin{align*}
    \vec r_{3q} &= \vec 0, & \vec r_{4q} &= a\,\hat\imath = 0{,}04\,\hat\imath,\\
    \vec r_{2q} &= a\,\hat\jmath = 0{,}04\,\hat\jmath, & \vec r_{q} &= a\,\hat\imath + a\,\hat\jmath.
\end{align*}
Centro: $\vec r_C = \tfrac{a}{2}(\hat\imath+\hat\jmath) = 0{,}02\,\hat\imath + 0{,}02\,\hat\jmath$.
Todas las esquinas est\'an a la misma distancia
\[
    r = \frac{a}{\sqrt2} = 0{,}0283\ \text{m},\qquad \frac{k}{r^{3}} = 3{,}98\times10^{14}.
\]
\end{minipage}

\medskip

\noindent\textbf{Campo en el centro} (superposici\'on, cargas con signo). Como todas equidistan de $C$:
\[
    \vec E_C = k\sum_i \frac{q_i\,(\vec r_C-\vec r_i)}{\lVert\vec r_C-\vec r_i\rVert^{3}}
             = \frac{k}{r^{3}}\sum_i q_i\,(\vec r_C-\vec r_i).
\]
Restas de vectores (todas de norma $r$) y sus productos $q_i(\vec r_C-\vec r_i)$:
\begin{align*}
    3q:\ \ \vec r_C-\vec r_{3q} &= 0{,}02\,\hat\imath+0{,}02\,\hat\jmath, & (15\,\mu C)(\ldots) &= (3\,\hat\imath+3\,\hat\jmath)\times10^{-7},\\
    4q:\ \ \vec r_C-\vec r_{4q} &= -0{,}02\,\hat\imath+0{,}02\,\hat\jmath, & (20\,\mu C)(\ldots) &= (-4\,\hat\imath+4\,\hat\jmath)\times10^{-7},\\
    2q:\ \ \vec r_C-\vec r_{2q} &= 0{,}02\,\hat\imath-0{,}02\,\hat\jmath, & (10\,\mu C)(\ldots) &= (2\,\hat\imath-2\,\hat\jmath)\times10^{-7},\\
    q:\ \ \vec r_C-\vec r_{q} &= -0{,}02\,\hat\imath-0{,}02\,\hat\jmath, & (5\,\mu C)(\ldots) &= (-1\,\hat\imath-1\,\hat\jmath)\times10^{-7}.
\end{align*}
Sumando (la componente $\hat\imath$ se cancela por simetr\'ia: lados izq.\ $2q{+}3q=5q$ = der.\ $q{+}4q=5q$):
\[
    \sum_i q_i(\vec r_C-\vec r_i) = (0)\,\hat\imath + (4\times10^{-7})\,\hat\jmath \quad[\text{C\,m}]
\]
\[
    \vec E_C = (3{,}98\times10^{14})(4\times10^{-7})\,\hat\jmath
    \quad\Longrightarrow\quad
    \boxed{\vec E_C = 1{,}59\times10^{8}\,\hat\jmath\ [\text{N/C}]}
\]
El campo apunta \textbf{hacia arriba} ($+\hat\jmath$): las cargas inferiores ($3q+4q$) superan a las superiores ($2q+q$).

\medskip

\noindent\textbf{Fuerza sobre un electr\'on} ($q_{e}=-e=-1{,}6\times10^{-19}$ C):
\[
    \vec F_{e^-} = q_e\,\vec E_C = (-1{,}6\times10^{-19})(1{,}59\times10^{8})\,\hat\jmath
    \quad\Longrightarrow\quad
    \boxed{\vec F_{e^-} = -2{,}55\times10^{-11}\,\hat\jmath\ [\text{N}]}
\]
(hacia abajo, opuesta a $\vec E$).

\medskip

\noindent\textbf{Fuerza sobre un prot\'on} ($q_{p}=+e=+1{,}6\times10^{-19}$ C):
\[
    \vec F_{p^+} = q_p\,\vec E_C = (+1{,}6\times10^{-19})(1{,}59\times10^{8})\,\hat\jmath
    \quad\Longrightarrow\quad
    \boxed{\vec F_{p^+} = +2{,}55\times10^{-11}\,\hat\jmath\ [\text{N}]}
\]
(hacia arriba, en el sentido de $\vec E$).

\medskip

\noindent\textbf{Interpretaci\'on.} El campo en el centro es puramente vertical porque la carga total del
lado izquierdo ($2q+3q=5q$) iguala a la del derecho ($q+4q=5q$), anulando la componente horizontal.
Electr\'on y prot\'on sienten fuerzas de \emph{igual magnitud} ($2{,}55\times10^{-11}$ N) pero
\emph{sentidos opuestos}, por sus cargas de signo contrario.
