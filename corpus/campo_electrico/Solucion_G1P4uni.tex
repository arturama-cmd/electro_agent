{\centering\Large\textbf{Soluci\'on: Equilibrio de Carga en el Eje $X$}\par}
{\centering\normalsize Gu\'ia 1 -- Problema 4 Unidimensional\par}

\vspace{0.5cm}

%========================================
% ENUNCIADO
%========================================
\section*{Enunciado}

Dos cargas puntuales sobre el eje $X$: $q_1 = +2{,}5\ [\mu\text{C}]$ en $x=0$ y
$q_2 = -3{,}5\ [\mu\text{C}]$ en $x=0{,}6$~[m]. Encuentre el punto donde, al colocar una carga
$q_3 = +3\ [\mu\text{C}]$, la \textbf{fuerza neta} sobre ella sea nula.

\medskip
\noindent\textbf{M\'etodo.} Ley de Coulomb vectorial con cargas con signo,
\[
    \vec F_{3j} = K\,\frac{q_3\,q_j}{\lVert\vec r_3-\vec r_j\rVert^{3}}\,(\vec r_3-\vec r_j),
    \qquad K=9\times10^{9}\ \tfrac{\text{N m}^2}{\text{C}^2}.
\]
Posiciones de las fuentes: $\vec r_1=\vec 0$, $\vec r_2=0{,}6\,\hat\imath$~[m]. Coeficientes:
\[
    K\,q_3q_1 = 9\times10^9(3\times10^{-6})(2{,}5\times10^{-6}) = 0{,}0675,\qquad
    K\,q_3q_2 = 9\times10^9(3\times10^{-6})(-3{,}5\times10^{-6}) = -0{,}0945.
\]

\begin{itemize}\itemsep2pt
    \item \textbf{Regi\'on I} (izquierda de $q_1$): $q_1$ repele a $q_3$ hacia $-\hat\imath$ y $q_2$ lo atrae hacia $+\hat\imath$: \emph{opuestas}, puede haber equilibrio. \checkmark
    \item \textbf{Regi\'on II} (entre las cargas): $q_1$ repele hacia $+\hat\imath$ y $q_2$ atrae hacia $+\hat\imath$: \emph{paralelas}, no hay equilibrio. $\times$
    \item \textbf{Regi\'on III} (derecha de $q_2$): fuerzas opuestas; se verifica algebraicamente.
\end{itemize}

%========================================
% REGION I
%========================================
\section*{Regi\'on I: $q_3$ a la izquierda de $q_1$}

Con $x>0$ la distancia a $q_1$, la posici\'on de la carga de prueba es $\vec r_3 = x(-\hat\imath)=-x\,\hat\imath$.

\noindent\textbf{Restas de vectores:}
\begin{align*}
    \vec r_3-\vec r_1 &= -x\,\hat\imath-\vec 0 = -x\,\hat\imath, & \lVert\vec r_3-\vec r_1\rVert &= x,\\
    \vec r_3-\vec r_2 &= -x\,\hat\imath-0{,}6\,\hat\imath = -(0{,}6+x)\,\hat\imath, & \lVert\vec r_3-\vec r_2\rVert &= 0{,}6+x.
\end{align*}
\noindent\textbf{Fuerzas} (el signo de $q_3q_j$ da el sentido):
\begin{align*}
    \vec F_{31} &= \frac{0{,}0675}{x^{3}}\,(-x\,\hat\imath) = -\frac{0{,}0675}{x^{2}}\,\hat\imath
        &&(\text{repulsi\'on},\ -\hat\imath),\\[2pt]
    \vec F_{32} &= \frac{-0{,}0945}{(0{,}6+x)^{3}}\,\big[-(0{,}6+x)\hat\imath\big] = \frac{0{,}0945}{(0{,}6+x)^{2}}\,\hat\imath
        &&(\text{atracci\'on},\ +\hat\imath).
\end{align*}
\noindent\textbf{Equilibrio} $\vec F_{31}+\vec F_{32}=\vec 0$:
\[
    \left[\frac{0{,}0945}{(0{,}6+x)^{2}}-\frac{0{,}0675}{x^{2}}\right]\hat\imath = \vec 0
    \quad\Longrightarrow\quad
    \frac{0{,}0675}{x^{2}} = \frac{0{,}0945}{(0{,}6+x)^{2}}
\]
\[
    0{,}0945\,x^{2} = 0{,}0675\,(x^{2}+1{,}2x+0{,}36)
    \quad\Longrightarrow\quad
    \boxed{0{,}027\,x^{2} - 0{,}081\,x - 0{,}0243 = 0}
\]
Con la f\'ormula cuadr\'atica:
\[
    x = \frac{0{,}081 \pm \sqrt{(0{,}081)^2+4(0{,}027)(0{,}0243)}}{2(0{,}027)}
      = \frac{0{,}081 \pm 0{,}0958}{0{,}054}
\]
\[
    x_1 = 3{,}275\ [\text{m}]\ \checkmark \qquad
    x_2 = -0{,}275\ [\text{m}]\ \times\ (\text{se descarta}, x>0)
\]

%========================================
% REGION III
%========================================
\section*{Regi\'on III: $q_3$ a la derecha de $q_2$}

Con $x>0$ la distancia a $q_2$, la posici\'on es $\vec r_3=(0{,}6+x)\,\hat\imath$.
\begin{align*}
    \vec r_3-\vec r_1 &= (0{,}6+x)\,\hat\imath, & \lVert\cdot\rVert &= 0{,}6+x,\\
    \vec r_3-\vec r_2 &= x\,\hat\imath, & \lVert\cdot\rVert &= x.
\end{align*}
\[
    \vec F_{31} = \frac{0{,}0675}{(0{,}6+x)^{2}}\,\hat\imath\ (+\hat\imath),\qquad
    \vec F_{32} = \frac{-0{,}0945}{x^{3}}\,(x\,\hat\imath) = -\frac{0{,}0945}{x^{2}}\,\hat\imath\ (-\hat\imath).
\]
Equilibrio $\Rightarrow \dfrac{0{,}0675}{(0{,}6+x)^{2}}=\dfrac{0{,}0945}{x^{2}}$, que conduce a
\[
    0{,}027\,x^{2} + 0{,}1134\,x + 0{,}03402 = 0
    \quad\Longrightarrow\quad
    x_1 = -0{,}325,\ \ x_2 = -3{,}875\ [\text{m}].
\]
Ambas ra\'ices son negativas $\Rightarrow$ \textbf{no hay soluci\'on f\'isica en la Regi\'on III}. $\times$

%========================================
% RESULTADO
%========================================
\section*{Resultado Final}

\[
    \boxed{\ \vec F_{\text{neta}}=\vec 0\ \text{a } x = 3{,}275\ [\text{m}]\ \text{a la izquierda de } q_1\ }
\]

\section*{Interpretaci\'on F\'isica}

\begin{itemize}\itemsep2pt
    \item El punto nulo aparece \emph{fuera} del par y del \textbf{lado de la carga m\'as d\'ebil} ($q_1$, $+2{,}5\,\mu$C): all\'i la repulsi\'on de $q_1$ (cercana) puede igualar la atracci\'on de $q_2$ (m\'as intensa pero lejana).
    \item En la Regi\'on III no hay soluci\'on porque, estando m\'as cerca de $q_2$ (la carga mayor, $|q_2|>|q_1|$), su atracci\'on siempre supera la repulsi\'on de $q_1$.
    \item Hallar la fuerza neta nula sobre $q_3$ equivale a buscar el punto de \textbf{campo el\'ectrico nulo} $\vec E=\vec 0$ del sistema.
\end{itemize}
