se ubican seg\'un la Fig.\ 1 (coordenadas en cm). Determine la fuerza el\'ectrica \textbf{resultante

\textbf{Vectores de posici\'on} (en metros; $q_2$ a $12$ cm y $40^\circ$ en el 2\textsuperscript{o} cuadrante):
\begin{align*}
    \vec r_1 &= 0{,}12\,\hat\imath - 0{,}01\,\hat\jmath,\\
    \vec r_2 &= -0{,}12\cos40^\circ\,\hat\imath + 0{,}12\,\mathrm{sen}\,40^\circ\,\hat\jmath
             \approx -0{,}09\,\hat\imath + 0{,}08\,\hat\jmath,\\
    \vec r_3 &= -0{,}11\,\hat\imath - 0{,}10\,\hat\jmath.
\end{align*}
Por superposici\'on, $\vec F_2 = \vec F_{21} + \vec F_{23}$, con
\[
    \vec F_{2j} = k\,\frac{q_2\,q_j}{\lVert\vec r_2-\vec r_j\rVert^3}\,(\vec r_2-\vec r_j).
\]
\end{minipage}

\medskip

\noindent\textbf{Restas de vectores:}
\begin{align*}
    \vec r_2-\vec r_1 &= (-0{,}09\,\hat\imath + 0{,}08\,\hat\jmath) - (0{,}12\,\hat\imath - 0{,}01\,\hat\jmath)
        = -0{,}21\,\hat\imath + 0{,}09\,\hat\jmath, & \lVert\vec r_2-\vec r_1\rVert &= \sqrt{0{,}21^2+0{,}09^2} \approx 0{,}23\ \text{m},\\[2pt]
    \vec r_2-\vec r_3 &= (-0{,}09\,\hat\imath + 0{,}08\,\hat\jmath) - (-0{,}11\,\hat\imath - 0{,}10\,\hat\jmath)
        = 0{,}02\,\hat\imath + 0{,}18\,\hat\jmath, & \lVert\vec r_2-\vec r_3\rVert &= \sqrt{0{,}02^2+0{,}18^2} \approx 0{,}18\ \text{m}.
\end{align*}

\noindent\textbf{Fuerza de $q_1$ sobre $q_2$} ($k\,q_2q_1 = 9\times10^9(-4\times10^{-6})(-5\times10^{-6}) = 0{,}18$):
\[
    \vec F_{21} = \frac{0{,}18}{(0{,}23)^3}\,(-0{,}21\,\hat\imath + 0{,}09\,\hat\jmath)
    = -3{,}11\,\hat\imath + 1{,}33\,\hat\jmath\ [\text{N}]
\]

\noindent\textbf{Fuerza de $q_3$ sobre $q_2$} ($k\,q_2q_3 = 9\times10^9(-4\times10^{-6})(3\times10^{-6}) = -0{,}108$):
\[
    \vec F_{23} = \frac{-0{,}108}{(0{,}18)^3}\,(0{,}02\,\hat\imath + 0{,}18\,\hat\jmath)
    = -0{,}37\,\hat\imath - 3{,}33\,\hat\jmath\ [\text{N}]
\]

\noindent\textbf{Fuerza resultante} (superposici\'on):
\begin{align*}
    F_x &= -3{,}11 - 0{,}37 = -3{,}48\ [\text{N}],\\[2pt]
    F_y &= \phantom{-}1{,}33 - 3{,}33 = -2{,}00\ [\text{N}].
\end{align*}
\[
    \boxed{\vec F_2 = -3{,}48\,\hat\imath - 2{,}00\,\hat\jmath\ [\text{N}],
    \qquad |\vec F_2| = \sqrt{3{,}48^2+2{,}00^2} = 4{,}02\ [\text{N}]}
\]

\noindent La resultante apunta al \textbf{tercer cuadrante} (abajo-izquierda), formando
$\arctan\!\big(\tfrac{2{,}00}{3{,}48}\big)=29{,}9^\circ$ por debajo del eje $-x$. Coherente con la
f\'isica: $\vec F_{21}$ es \textbf{repulsiva} ($q_1,q_2$ ambas negativas, empuja a $q_2$ alej\'andolo de
$q_1$: arriba-izquierda) y $\vec F_{23}$ es \textbf{atractiva} ($q_3$ positiva atrae a $q_2$: hacia
abajo, en direcci\'on a $q_3$).
