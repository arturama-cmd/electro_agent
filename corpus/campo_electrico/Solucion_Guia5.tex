\problema{Problema 1. Superposici\'on: plano, hilo y carga puntual}

\noindent Un \textbf{plano infinito} con densidad superficial $\sigma=-3\times10^{-6}\ \mathrm{C/m^2}$
(perpendicular al eje $x$, en $x=0$), un \textbf{hilo infinito} con densidad lineal
$\lambda=4\times10^{-4}\ \mathrm{C/m}$ (paralelo al eje $y$, en $x=10$ m) y una \textbf{carga puntual}
$q=5\times10^{-9}$ C en $\vec r_q=(4\,\hat\imath+6\,\hat\jmath)$ m. Determine el campo el\'ectrico neto
en el punto $P=(5\,\hat\imath+7\,\hat\jmath)$ m.

\medskip
\begin{center}
\begin{tikzpicture}[scale=0.72, >=Stealth, line width=0.8pt]
    % ejes
    \draw[->,gray] (-1,0)--(11.6,0) node[right,font=\small]{$x$ [m]};
    \draw[->,gray] (0,-0.6)--(0,8.2) node[above,font=\small]{$y$ [m]};
    \node[gray,font=\scriptsize,below left] at (0,0){$O$};
    % plano (sigma<0) en x=0 -> linea roja vertical
    \draw[cargarojo,line width=2.2pt] (0,-0.4)--(0,8);
    \foreach \yy in {0.4,1.4,2.4,3.4,4.4,5.4,6.4,7.4}{\node[cargarojo,font=\tiny] at (-0.28,\yy){$-$};}
    \node[cargarojo,font=\small] at (-0.75,4.2){$\sigma$};
    % hilo (lambda>0) en x=10 -> linea azul vertical
    \draw[cargazul,line width=2.2pt] (10,-0.4)--(10,8);
    \foreach \yy in {0.4,1.4,2.4,3.4,4.4,5.4,6.4,7.4}{\node[cargazul,font=\tiny] at (10.28,\yy){$+$};}
    \node[cargazul,font=\small] at (10.6,4.2){$\lambda$};
    % carga puntual (positiva) azul en (4,6)
    \filldraw[cargazul] (4,6) circle (3pt);
    \node[cargazul,font=\scriptsize,below] at (4,5.7){$q\,(4,6)$};
    % punto P en (5,7)
    \filldraw[black] (5,7) circle (3pt);
    \node[font=\scriptsize,above right] at (5,7.15){$P\,(5,7)$};
    % vectores de campo en P (esquem\'aticos): plano e hilo van en -i, carga en (i+j)
    \draw[->,cargarojo] (5,7)--(3.4,7) node[midway,above,font=\tiny]{$\vec E_\sigma$};
    \draw[->,cargazul] (5,6.55)--(3.4,6.55) node[midway,below,font=\tiny]{$\vec E_\lambda$};
    \draw[->,black] (5,7)--(6.15,8.05) node[right,font=\tiny]{$\vec E_q$};
    % cota r=5 entre P y el hilo
    \draw[dashed,gray] (5,7)--(10,7);
    \node[gray,font=\tiny] at (7.7,7.3){$r=5$ m};
\end{tikzpicture}\\[1pt]
{\scriptsize Figura 1. Configuraci\'on (no a escala). Superficies de Gauss: caja para el plano, cilindro para el hilo.}
\end{center}

\secgap

\noindent\textbf{(a) Campo del plano infinito.} Con una caja gaussiana que atraviesa el plano
($\sigma\equiv dq/dA$), la ley de Gauss $\oint\vec E\cdot d\vec S=Q_{enc}/\varepsilon_0$ da
\[
    E\,A=\frac{|\sigma|\,A}{\varepsilon_0}\quad\Longrightarrow\quad E_\sigma=\frac{|\sigma|}{\varepsilon_0}
    =\frac{3\times10^{-6}}{8{,}85\times10^{-12}}=3{,}39\times10^{5}\ \tfrac{\text{N}}{\text{C}}.
\]
Como $\sigma<0$, el campo apunta \emph{hacia} el plano; en $P$ (a la derecha del plano) va en $-\hat\imath$:
\[
    \boxed{\ \vec E_\sigma=3{,}39\times10^{5}\,(-\hat\imath)\ \tfrac{\text{N}}{\text{C}}\ }
\]

\secgap

\noindent\textbf{(b) Campo del hilo infinito.} Con un cilindro gaussiano de radio $r$ y largo $L$
($\lambda\equiv dq/d\ell$), el flujo s\'olo cruza el manto:
\[
    E\,(2\pi r L)=\frac{\lambda L}{\varepsilon_0}\quad\Longrightarrow\quad
    E_\lambda=\frac{\lambda}{2\pi\varepsilon_0 r}=\frac{7{,}19\times10^{6}}{r}.
\]
En $P$ la distancia al hilo es $r=5$ m y, como $\lambda>0$ y el hilo est\'a a la derecha, apunta en $-\hat\imath$:
\[
    E_\lambda=\frac{7{,}19\times10^{6}}{5}=1{,}44\times10^{6}\ \tfrac{\text{N}}{\text{C}}
    \quad\Longrightarrow\quad
    \boxed{\ \vec E_\lambda=1{,}44\times10^{6}\,(-\hat\imath)\ \tfrac{\text{N}}{\text{C}}\ }
\]

\secgap

\noindent\textbf{(c) Campo de la carga puntual.} Con la ley de Coulomb vectorial y
$\vec r_P-\vec r_q=(5-4)\hat\imath+(7-6)\hat\jmath=\hat\imath+\hat\jmath$,
$\ \lVert\vec r_P-\vec r_q\rVert=\sqrt{2}$ m:
\[
    \vec E_q=k\,\frac{q\,(\vec r_P-\vec r_q)}{\lVert\vec r_P-\vec r_q\rVert^{3}}
    =\frac{(9\times10^{9})(5\times10^{-9})}{(\sqrt2)^{3}}(\hat\imath+\hat\jmath)
    =\boxed{15{,}91\,(\hat\imath+\hat\jmath)\ \tfrac{\text{N}}{\text{C}}}
\]

\secgap

\noindent\textbf{(d) Campo neto (superposici\'on).}
\begin{align*}
    \vec E_T&=\vec E_\sigma+\vec E_\lambda+\vec E_q
    =3{,}39\times10^{5}(-\hat\imath)+1{,}44\times10^{6}(-\hat\imath)+15{,}91(\hat\imath+\hat\jmath),\\[2pt]
    &=\big(-1{,}78\times10^{6}\big)\hat\imath+15{,}91\,\hat\jmath.
\end{align*}
\[
    \boxed{\ \vec E_T=1{,}78\times10^{6}\,(-\hat\imath)+15{,}91\,\hat\jmath\ \tfrac{\text{N}}{\text{C}}\ }
\]
El plano y el hilo dominan (componente $-\hat\imath$); la carga puntual s\'olo aporta la peque\~na
componente $\hat\jmath$.

\secgap

\problema{Problema 3. Esfera maciza (Ley de Gauss)}

\noindent
\begin{minipage}[c]{0.32\textwidth}
\centering
\begin{tikzpicture}[scale=1.05, >=Stealth, line width=0.8pt]
    % esfera de radio a (relleno tenue)
    \fill[cargazul!12] (0,0) circle (1.7);
    \draw[cargazul,line width=1.2pt] (0,0) circle (1.7);
    % superficie gaussiana interior r1<a
    \draw[dashed,cargarojo] (0,0) circle (1.0);
    % superficie gaussiana exterior r2>a
    \draw[dashed,cargarojo] (0,0) circle (2.35);
    % centro y radios
    \fill (0,0) circle (1.3pt); \node[below left,font=\scriptsize] at (0,0){$O$};
    \draw[->] (0,0)--(35:1.0) node[midway,above,font=\scriptsize]{$r_1$};
    \draw[->] (0,0)--(-58:2.35) node[midway,below right,font=\scriptsize]{$r_2$};
    \draw[->,gray] (0,0)--(150:1.7) node[midway,above,font=\scriptsize]{$a$};
    % E radial
    \draw[->,cargarojo] (10:2.35)--(10:2.85);
    \node[cargazul,font=\scriptsize] at (0.55,-0.55){$\rho$};
\end{tikzpicture}\\[2pt]
{\scriptsize Figura 2. Esfera y superficies de Gauss ($r_1<a$, $r_2>a$).}
\end{minipage}%
\hfill
\begin{minipage}[c]{0.63\textwidth}
Esfera de radio $a$ con densidad volum\'etrica $\rho\equiv dq/dV$. Por simetr\'ia esf\'erica
$\vec E\parallel d\vec S$ y $E$ es constante sobre cada superficie gaussiana, de modo que
\[
    \oint\vec E\cdot d\vec S=E\,(4\pi r^{2})=\frac{Q_{enc}}{\varepsilon_0}.
\]
El campo queda determinado por la carga encerrada $Q_{enc}$ en cada regi\'on.
\end{minipage}

\secgap

\noindent\textbf{(a) y (b) Densidad uniforme} $\rho=\dfrac{Q}{\tfrac{4}{3}\pi a^{3}}$.

\smallskip
\noindent\emph{Interior} ($r_1<a$): $\ Q_{enc}=\rho\,\tfrac{4}{3}\pi r_1^{3}$, luego
\[
    E\,(4\pi r_1^{2})=\frac{\rho\,\tfrac{4}{3}\pi r_1^{3}}{\varepsilon_0}
    \ \Longrightarrow\
    E=\frac{\rho\,r_1}{3\varepsilon_0}
    \ \Longrightarrow\
    \boxed{\ E(r_1)=\frac{Q\,r_1}{4\pi\varepsilon_0 a^{3}}\ }\quad(r_1<a).
\]
\noindent\emph{Exterior} ($r_2>a$): $\ Q_{enc}=Q$, luego
\[
    E\,(4\pi r_2^{2})=\frac{Q}{\varepsilon_0}
    \ \Longrightarrow\
    \boxed{\ E(r_2)=\frac{Q}{4\pi\varepsilon_0 r_2^{2}}\ }\quad(r_2>a).
\]

\secgap

\noindent\textbf{(c) Densidad variable} $\rho=\rho_0\,\dfrac{r}{a}$. Como $\rho$ no es constante, se integra
$Q_{enc}=\displaystyle\int\rho\,dV$ con $dV=4\pi r^{2}\,dr$.

\smallskip
\noindent\emph{Interior} ($r_1<a$):
\begin{align*}
    Q_{enc}&=\int_0^{r_1}\rho_0\frac{r}{a}\,(4\pi r^{2})\,dr
    =\frac{4\pi\rho_0}{a}\int_0^{r_1} r^{3}\,dr=\frac{\pi\rho_0}{a}\,r_1^{4},\\[2pt]
    E\,(4\pi r_1^{2})&=\frac{1}{\varepsilon_0}\frac{\pi\rho_0}{a}r_1^{4}
    \ \Longrightarrow\
    \boxed{\ E(r_1)=\frac{\rho_0\,r_1^{2}}{4\,a\,\varepsilon_0}\ }\quad(r_1<a).
\end{align*}
\noindent\emph{Exterior} ($r_2>a$): la carga s\'olo existe hasta $r=a$, por lo que
\begin{align*}
    Q_{enc}&=\int_0^{a}\rho_0\frac{r}{a}\,(4\pi r^{2})\,dr=\frac{4\pi\rho_0}{a}\frac{a^{4}}{4}=\pi\rho_0 a^{3},\\[2pt]
    E\,(4\pi r_2^{2})&=\frac{\pi\rho_0 a^{3}}{\varepsilon_0}
    \ \Longrightarrow\
    \boxed{\ E(r_2)=\frac{\rho_0\,a^{3}}{4\,\varepsilon_0\,r_2^{2}}\ }\quad(r_2>a).
\end{align*}
En ambos casos, para $r>a$ el campo decae como $1/r^{2}$ (como el de una carga puntual con la carga total).

\secgap

\problema{Problema 7. Dos hilos paralelos: punto de campo nulo}

\noindent Dos hilos infinitos paralelos separados una distancia $L$: el hilo 1 con
$\lambda_1=6\times10^{-6}$ C/m ($>0$) y el hilo 2 con $\lambda_2=-2\times10^{-6}$ C/m ($<0$).
Determine el punto sobre el eje donde el campo el\'ectrico neto es \textbf{cero}.

\medskip
\begin{center}
\begin{tikzpicture}[scale=1.0, >=Stealth, line width=0.8pt]
    % eje
    \draw[->,gray] (-3.2,0)--(7.2,0) node[right,font=\small]{eje};
    % hilo 1 (lambda1>0) azul en x=0
    \draw[cargazul,line width=2.4pt] (0,-1.3)--(0,1.3);
    \node[cargazul,font=\small,above] at (0,1.35){$\lambda_1$};
    \draw[dashed,cargarojo] (0,0) ellipse (0.45 and 1.0);
    % hilo 2 (lambda2<0) rojo en x=4 (=L)
    \draw[cargarojo,line width=2.4pt] (4,-1.3)--(4,1.3);
    \node[cargarojo,font=\small,above] at (4,1.35){$\lambda_2$};
    \draw[dashed,cargarojo] (4,0) ellipse (0.45 and 1.0);
    % separacion L
    \draw[<->,gray] (0,-1.6)--(4,-1.6) node[midway,below,font=\scriptsize]{$L$};
    % regiones
    \node[font=\scriptsize] at (-2.2,0.35){I};
    \node[font=\scriptsize] at (2.0,0.35){II};
    \node[font=\scriptsize] at (5.1,0.7){III};
    % punto de campo nulo en region III (x2 = L/2 a la derecha de lambda2)
    \filldraw[black] (6,0) circle (2.6pt);
    \node[font=\scriptsize,above] at (6,0.15){$P_0$};
    \draw[<->,gray] (4,-0.55)--(6,-0.55) node[midway,below,font=\scriptsize]{$L/2$};
\end{tikzpicture}\\[1pt]
{\scriptsize Figura 3. Hilos paralelos y regiones I, II, III (cilindros de Gauss en rojo punteado).}
\end{center}

\secgap

\noindent\textbf{Campo de cada hilo (Gauss).} $E_i=\dfrac{|\lambda_i|}{2\pi\varepsilon_0\,r_i}$, radial a cada hilo.

\smallskip
\noindent\emph{Regi\'on II (entre los hilos):} $\vec E_1$ y $\vec E_2$ apuntan en el \emph{mismo} sentido
(el hilo positivo repele y el negativo atrae hacia s\'i), por lo que \textbf{no se anulan}.

\smallskip
\noindent\emph{Regiones I y III (fuera):} $\vec E_1$ y $\vec E_2$ se oponen. La condici\'on $\vec E=\vec 0$ pide
\[
    \frac{\lambda_1}{2\pi\varepsilon_0 r_1}=\frac{|\lambda_2|}{2\pi\varepsilon_0 r_2}
    \ \Longrightarrow\
    \frac{r_1}{r_2}=\frac{\lambda_1}{|\lambda_2|}=\frac{6\times10^{-6}}{2\times10^{-6}}=3
    \ \Longrightarrow\
    r_1=3\,r_2.
\]

\smallskip
\noindent\emph{Regi\'on III (a la derecha de ambos):} $r_1=r_2+L$. Sustituyendo $r_1=3r_2$:
\[
    r_2+L=3r_2\ \Longrightarrow\ 2r_2=L\ \Longrightarrow\
    \boxed{\ r_2=\frac{L}{2}\ }\quad(\text{a }L/2\text{ a la derecha de }\lambda_2).
\]

\smallskip
\noindent\emph{Regi\'on I (a la izquierda de ambos):} $r_2=r_1+L$, y $r_1=3r_2$ da $r_1=3(r_1+L)$,
es decir $r_1=-\tfrac{3}{2}L<0$: \textbf{no hay soluci\'on} (las distancias deben ser positivas).

\smallskip
\noindent Por lo tanto, el campo neto se anula en un \'unico punto, en la \textbf{regi\'on III}, a una
distancia $L/2$ del hilo $\lambda_2$ (equivalentemente, $3L/2$ del hilo $\lambda_1$).

\secgap
