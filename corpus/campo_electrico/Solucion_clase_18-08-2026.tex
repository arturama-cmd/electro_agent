\noindent Por superposici\'on $\vec F_2 = \vec F_{21}+\vec F_{23}+\vec F_{24}$, con
$\vec F_{2j}=k\dfrac{q_2 q_j}{\lVert\vec r_2-\vec r_j\rVert^{3}}(\vec r_2-\vec r_j)$.
Coeficientes: $kq_2q_1=-1{,}8\times10^{-7}$, $kq_2q_3=-2{,}52\times10^{-7}$, $kq_2q_4=+3{,}6\times10^{-7}$.

\medskip
\noindent\textbf{Restas de vectores} ($\vec r_2-\vec r_j$):
\begin{align*}
    \vec r_2-\vec r_1 &= -0{,}274\hat\imath-0{,}324\hat\jmath, & \lVert\cdot\rVert&=0{,}424,\\
    \vec r_2-\vec r_3 &= -0{,}390\hat\imath-0{,}160\hat\jmath, & \lVert\cdot\rVert&=0{,}422,\\
    \vec r_2-\vec r_4 &= -0{,}160\hat\imath-0{,}350\hat\jmath, & \lVert\cdot\rVert&=0{,}385.
\end{align*}
\begin{align*}
    \vec F_{21} &= \tfrac{-1{,}8\times10^{-7}}{(0{,}424)^3}(-0{,}274\hat\imath-0{,}324\hat\jmath) = 6{,}470\times10^{-7}\hat\imath+7{,}651\times10^{-7}\hat\jmath\ [\text{N}],\\
    \vec F_{23} &= \tfrac{-2{,}52\times10^{-7}}{(0{,}422)^3}(-0{,}390\hat\imath-0{,}160\hat\jmath) = 1{,}308\times10^{-6}\hat\imath+5{,}365\times10^{-7}\hat\jmath\ [\text{N}],\\
    \vec F_{24} &= \tfrac{+3{,}6\times10^{-7}}{(0{,}385)^3}(-0{,}160\hat\imath-0{,}350\hat\jmath) = -1{,}010\times10^{-6}\hat\imath-2{,}208\times10^{-6}\hat\jmath\ [\text{N}].
\end{align*}
\[
    \boxed{\vec F_2 = 9{,}45\times10^{-7}\,\hat\imath - 9{,}064\times10^{-7}\,\hat\jmath\ [\text{N}],\qquad
    |\vec F_2| = 1{,}31\times10^{-6}\ [\text{N}]}
\]

\section*{Problema 2: Campo el\'ectrico resultante en el punto $P$}

\noindent Con las mismas cuatro cargas (en el plano $z=0$), determine el campo $\vec E$ en el punto
$P=(-35,-40,8)$ cm, es decir $\vec r_P = -0{,}35\hat\imath-0{,}40\hat\jmath+0{,}08\hat k$ m.

\begin{center}
\begin{tikzpicture}[scale=0.9, >=Stealth, line width=0.7pt,
    x={(-0.5cm,-0.35cm)}, y={(1cm,0cm)}, z={(0cm,1cm)}]
    \draw[->,gray] (0,0,0)--(3.2,0,0) node[below,black,font=\small]{$x$};
    \draw[->,gray] (0,0,0)--(0,3.4,0) node[right,black,font=\small]{$y$};
    \draw[->,gray] (0,0,0)--(0,0,2.6) node[above,black,font=\small]{$z$};
    % plano z=0: cargas (esquematico, no a escala)
    \filldraw[orange!80!black] (0.5,0.9,0) circle (2pt) node[right,font=\scriptsize]{$q_1$};
    \filldraw[orange!80!black] (1.6,-0.8,0) circle (2pt) node[right,font=\scriptsize]{$q_2$};
    \filldraw[orange!80!black] (-0.7,1.3,0) circle (2pt) node[right,font=\scriptsize]{$q_3$};
    \filldraw[orange!80!black] (1.1,-0.3,0) circle (2pt) node[left,font=\scriptsize]{$q_4$};
    % punto P (fuera del plano, z=8)
    \filldraw[red] (2.5,-2.4,0.8) circle (2.4pt) node[right,red,font=\scriptsize]{$P$};
    \draw[dashed,red!50] (2.5,-2.4,0)--(2.5,-2.4,0.8);
    \node[font=\scriptsize,gray] at (0,0,3.0){[cm]};
\end{tikzpicture}
\end{center}

\noindent $\vec E = \sum_j \vec E_j$, con $\vec E_j=k\dfrac{q_j}{\lVert\vec r_P-\vec r_j\rVert^{3}}(\vec r_P-\vec r_j)$.
Coeficientes $kq_j$: $\ 45,\ -36,\ 63,\ -90$ (para $q_1,q_2,q_3,q_4$).

\medskip
\noindent\textbf{Restas de vectores} ($\vec r_P-\vec r_j$, con componente $\hat k=0{,}08$):
\begin{align*}
    \vec r_P-\vec r_1 &= -0{,}404\hat\imath-0{,}444\hat\jmath+0{,}08\hat k, & \lVert\cdot\rVert&=0{,}606,\\
    \vec r_P-\vec r_2 &= -0{,}130\hat\imath-0{,}120\hat\jmath+0{,}08\hat k, & \lVert\cdot\rVert&=0{,}194,\\
    \vec r_P-\vec r_3 &= -0{,}520\hat\imath-0{,}280\hat\jmath+0{,}08\hat k, & \lVert\cdot\rVert&=0{,}596,\\
    \vec r_P-\vec r_4 &= -0{,}290\hat\imath-0{,}470\hat\jmath+0{,}08\hat k, & \lVert\cdot\rVert&=0{,}558.
\end{align*}
\begin{align*}
    \vec E_1 &= \tfrac{45}{(0{,}606)^3}(-0{,}404\hat\imath-0{,}444\hat\jmath+0{,}08\hat k) = -81{,}69\hat\imath-89{,}78\hat\jmath+16{,}18\hat k,\\
    \vec E_2 &= \tfrac{-36}{(0{,}194)^3}(-0{,}130\hat\imath-0{,}120\hat\jmath+0{,}08\hat k) = 640{,}97\hat\imath+591{,}67\hat\jmath-394{,}45\hat k,\\
    \vec E_3 &= \tfrac{63}{(0{,}596)^3}(-0{,}520\hat\imath-0{,}280\hat\jmath+0{,}08\hat k) = -154{,}7\hat\imath-83{,}3\hat\jmath+23{,}8\hat k,\\
    \vec E_4 &= \tfrac{-90}{(0{,}558)^3}(-0{,}290\hat\imath-0{,}470\hat\jmath+0{,}08\hat k) = 150{,}2\hat\imath+243{,}5\hat\jmath-41{,}4\hat k.
\end{align*}
Sumando componente a componente ($\hat\imath$: $-81{,}69+640{,}97-154{,}7+150{,}2$; etc.):
\[
    \boxed{\vec E_P = 554{,}8\,\hat\imath + 662{,}1\,\hat\jmath - 395{,}9\,\hat k\ [\text{N/C}],\qquad
    |\vec E_P| = 950\ [\text{N/C}]}
\]

\noindent\textbf{Interpretaci\'on.} En ambos problemas se usa la misma configuraci\'on de cargas y el
formalismo vectorial con cargas con signo. En el problema 2 el punto $P$ est\'a \emph{fuera} del plano
de las cargas ($z=8$ cm), por lo que el campo tiene componente $\hat k$; domina $\vec E_2$ por ser $q_2$
la carga m\'as cercana a $P$ ($\lVert\vec r_P-\vec r_2\rVert=0{,}194$ m, la menor distancia).
