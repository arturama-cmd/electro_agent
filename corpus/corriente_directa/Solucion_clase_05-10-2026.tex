\section*{Problema 1 -- $C_1$ con dos diel\'ectricos en serie, en paralelo con $C_2$}

\noindent\textbf{Enunciado:}
Un condensador de placas paralelas $C_1$ tiene \'area $A = 30\ [\text{cm}^2]$ y est\'a relleno con
\textbf{dos l\'aminas diel\'ectricas en serie}, cada una de espesor $d = 1{,}00\ [\text{mm}]$, con
constantes $\kappa_1 = 2$ y $\kappa_2 = 6$. \'Este se conecta en \textbf{paralelo} con
$C_2 = 15\ [\text{pF}]$ a una fuente $V = 24\ [\text{V}]$. Determine:

\begin{enumerate}[(a)]
    \item La capacitancia equivalente del circuito.
    \item La energ\'ia almacenada en cada condensador.
\end{enumerate}

\medskip
\noindent\textbf{Datos:}
\[
    A = 30\ [\text{cm}^2] = 3{,}0\times10^{-3}\ [\text{m}^2],\quad
    d = 1{,}00\ [\text{mm}]\ (\text{cada capa}),\quad
    \kappa_1 = 2,\quad \kappa_2 = 6,
\]
\[
    C_2 = 15\ [\text{pF}],\quad V = 24\ [\text{V}],\quad
    \varepsilon_0 = 8{,}854\times10^{-12}\ [\text{F/m}]
\]

\begin{center}
\begin{tikzpicture}[scale=1.1]
    % Fuente V (izquierda)
    \draw[thick] (0,3.5) -- (0,0);
    \draw[very thick] (-0.28,1.90) -- (0.28,1.90);

{\footnotesize Fig.\ 1. $C_1$ con dos capas diel\'ectricas en serie, en paralelo con $C_2$.}
\end{center}

\medskip
\noindent\textbf{Parte (a) -- Capacitancia equivalente.}

Las dos l\'aminas diel\'ectricas dentro de $C_1$ (misma \'area $A$, mismo espesor $d$, distinta
$\kappa$) se comportan como dos condensadores en \textbf{serie}:
\begin{align*}
    C_{\kappa_1} &= \kappa_1\,\varepsilon_0\,\frac{A}{d}
                  = 2\times 8{,}854\times10^{-12}\times\frac{3{,}0\times10^{-3}}{1{,}00\times10^{-3}}
                  = 53{,}12\ [\text{pF}]\\[4pt]
    C_{\kappa_2} &= \kappa_2\,\varepsilon_0\,\frac{A}{d}
                  = 6\times 8{,}854\times10^{-12}\times 3{,}0
                  = 159{,}37\ [\text{pF}]
\end{align*}

$C_{\kappa_1}$ y $C_{\kappa_2}$ en serie forman $C_1$:
\begin{align*}
    \frac{1}{C_1} &= \frac{1}{C_{\kappa_1}} + \frac{1}{C_{\kappa_2}}
                   = \frac{1}{53{,}12} + \frac{1}{159{,}37}\ [\text{pF}^{-1}]\\[4pt]
    C_1           &= \varepsilon_0\,\frac{A}{d}\cdot\frac{\kappa_1\kappa_2}{\kappa_1+\kappa_2}
                   = 8{,}854\times10^{-12}\times 3{,}0\times\frac{12}{8}
                   = 39{,}84\ [\text{pF}]
\end{align*}

$C_1$ y $C_2$ en \textbf{paralelo}:
\[
    \boxed{C_{eq} = C_1 + C_2 = 39{,}84 + 15{,}0 = 54{,}84\ [\text{pF}]}
\]

\medskip
\noindent\textbf{Parte (b) -- Energ\'ia almacenada en cada condensador.}

Como $C_1 \parallel C_2$, ambos quedan sometidos al voltaje de la fuente:
$V_1 = V_2 = V = 24{,}0\ [\text{V}]$. La energ\'ia de cada uno es $U = \tfrac12\,C\,V^2$:
\begin{align*}
    U_1 &= \tfrac12\,C_1\,V^2
         = \tfrac12\times 39{,}84\times10^{-12}\times(24{,}0)^2
         = \tfrac12\times 39{,}84\times10^{-12}\times 576
\end{align*}
\[
    \boxed{U_1 = 1{,}147\times10^{-8}\ [\text{J}] = 11{,}47\ [\text{nJ}]}
\]
\begin{align*}
    U_2 &= \tfrac12\,C_2\,V^2
         = \tfrac12\times 15{,}0\times10^{-12}\times 576
\end{align*}
\[
    \boxed{U_2 = 4{,}32\times10^{-9}\ [\text{J}] = 4{,}32\ [\text{nJ}]}
\]

\noindent\textbf{Interpretaci\'on f\'isica:} aunque $C_1$ y $C_2$ comparten el mismo voltaje, $C_1$
almacena $\approx 2{,}7$ veces m\'as energ\'ia que $C_2$, pues a igual $V$ se cumple $U\propto C$
($C_1/C_2 = 39{,}84/15{,}0 \approx 2{,}66$). Dentro de $C_1$, las dos capas en serie producen una
capacitancia efectiva gobernada por el diel\'ectrico de \emph{menor} $\kappa$ (la capa
$\kappa_1=2$), de forma an\'aloga a resistencias en serie donde domina la mayor resistencia.

\section*{Problema 2 -- Circuito mixto de siete condensadores}

\begin{enumerate}[(a)]

\medskip
\noindent\textbf{Parte (a) -- Capacitancia equivalente.}

El condensador $C_7$ en la rama $b$--$d$ separa el nodo $b$ del \textbf{nodo central}
$M=\{c,d,e\}$ (los puntos $c$, $d$ y $e$ quedan unidos por los cables $c$--$e$ y $d$--$e$). Entonces:
\begin{itemize}
    \item Entre $b$ y $M$: $C_2$, $C_3$ y $C_7$ en \textbf{paralelo}.
    \item Entre $M$ y $g$: $C_4$ y $C_5$ en \textbf{paralelo}.
    \item $C_1$ ($a$--$b$), el grupo $b$--$M$, el grupo $M$--$g$ y $C_6$ ($f$--$g$) quedan en
          \textbf{serie} con la fuente.
\end{itemize}

\[
    C_{bM} = C_2 + C_3 + C_7 = 8{,}0 + 5{,}0 + 20{,}0 = 33{,}0\ [\mu\text{F}]
\]
\[
    C_{Mg} = C_4 + C_5 = 12{,}0 + 6{,}0 = 18{,}0\ [\mu\text{F}]
\]

Reducci\'on de los cuatro bloques en serie:
\begin{align*}
    \frac{1}{C_{eq}} &= \frac{1}{C_1} + \frac{1}{C_{bM}} + \frac{1}{C_{Mg}} + \frac{1}{C_6}\\[4pt]
                     &= \frac{1}{5{,}0} + \frac{1}{33{,}0} + \frac{1}{18{,}0} + \frac{1}{5{,}0}
                      = \frac{481}{990}\ [\mu\text{F}^{-1}]
\end{align*}
\[
    \boxed{C_{eq} = \frac{990}{481} = 2{,}058\ [\mu\text{F}]}
\]

\medskip
\noindent\textbf{Parte (b) -- Carga en cada condensador.}

Los cuatro bloques en serie ($C_1$, $C_{bM}$, $C_{Mg}$, $C_6$) comparten la \textbf{misma carga},
igual a la carga total entregada por la fuente:
\[
    Q = C_{eq}\,V = 2{,}058\times10^{-6}\times 12{,}0 = 24{,}70\ [\mu\text{C}]
\]

\noindent\emph{Condensadores simples en serie:} $C_1$ y $C_6$ llevan toda la carga $Q$:
\[
    \boxed{q_1 = q_6 = Q = 24{,}70\ [\mu\text{C}]}
\]

\noindent\emph{Grupo $b$--$M$} ($C_2\parallel C_3\parallel C_7$): mismo voltaje
$V_{bM} = Q/C_{bM} = 24{,}70/33{,}0 = 0{,}7484\ [\text{V}]$:
\begin{align*}
    q_2 &= C_2\,V_{bM} = 8{,}0\times 0{,}7484 = 5{,}99\ [\mu\text{C}]\\
    q_3 &= C_3\,V_{bM} = 5{,}0\times 0{,}7484 = 3{,}74\ [\mu\text{C}]\\
    q_7 &= C_7\,V_{bM} = 20{,}0\times 0{,}7484 = 14{,}97\ [\mu\text{C}]
\end{align*}

\noindent\emph{Grupo $M$--$g$} ($C_4\parallel C_5$): mismo voltaje
$V_{Mg} = Q/C_{Mg} = 24{,}70/18{,}0 = 1{,}372\ [\text{V}]$:
\begin{align*}
    q_4 &= C_4\,V_{Mg} = 12{,}0\times 1{,}372 = 16{,}47\ [\mu\text{C}]\\
    q_5 &= C_5\,V_{Mg} = 6{,}0\times 1{,}372 = 8{,}23\ [\mu\text{C}]
\end{align*}

\medskip
\noindent\textbf{Verificaci\'on:}
\[
    q_2+q_3+q_7 = 5{,}99+3{,}74+14{,}97 = 24{,}70\ [\mu\text{C}] = Q\ \checkmark
    \qquad
    q_4+q_5 = 16{,}47+8{,}23 = 24{,}70\ [\mu\text{C}] = Q\ \checkmark
\]
Voltajes en serie (con $V_1 = V_6 = Q/C_1 = 24{,}70/5{,}0 = 4{,}94\ [\text{V}]$):
\[
    V_1 + V_{bM} + V_{Mg} + V_6 = 4{,}94 + 0{,}748 + 1{,}372 + 4{,}94 = 12{,}0\ [\text{V}] = V\ \checkmark
\]
