\section*{Problema 1 -- Leyes de Kirchhoff (Gu\'ia 8, Problema 1)}

\subsection*{Enunciado}

En el circuito de la figura las fuentes son ideales (sin resistencia interna), con
$\varepsilon_1 = 20\ [\text{V}]$ y $\varepsilon_2 = 24\ [\text{V}]$, y las resistencias valen
$R_1 = 4\ [\Omega]$, $R_2 = 6\ [\Omega]$, $R_3 = 12\ [\Omega]$, $R_4 = 2\ [\Omega]$,
$R_5 = 3\ [\Omega]$ y $R_6 = 12\ [\Omega]$. Use las leyes de Kirchhoff para determinar la
corriente $I$ que circula por $R_5$, con el sentido de referencia indicado por la flecha
(hacia arriba). Un resultado negativo indicar\'ia que la corriente circula en sentido contrario.

\subsection*{Datos}

\[
    \varepsilon_1 = 20\ [\text{V}],\quad \varepsilon_2 = 24\ [\text{V}],\quad
    R_1 = 4\ [\Omega],\quad R_2 = 6\ [\Omega],\quad R_3 = 12\ [\Omega],
\]
\[
    R_4 = 2\ [\Omega],\quad R_5 = 3\ [\Omega],\quad R_6 = 12\ [\Omega]
\]

\subsection*{An\'alisis del circuito}

El circuito tiene \textbf{dos nodos}, $A$ (uni\'on superior) y $B$ (uni\'on inferior), unidos
por \textbf{tres ramas}. Los dem\'as puntos son simples esquinas, de modo que en cada rama
circula una sola corriente:
\begin{itemize}\setlength\itemsep{1pt}
    \item \textbf{Rama izquierda:} $R_1$, $\varepsilon_1$ y $R_5$ en serie. Corriente $I_1$, que
          sube por $R_5$ y llega a $A$. Es la corriente pedida: $I = I_1$.
    \item \textbf{Rama central:} $R_3$. Corriente $I_2$, de $A$ hacia $B$.
    \item \textbf{Rama derecha:} $R_2$, $R_4$, $\varepsilon_2$ y $R_6$ en serie. Corriente $I_3$,
          que sale de $A$ por $R_2$ y llega a $B$ por $R_6$.
\end{itemize}
La fuente $\varepsilon_1$ tiene su borne $+$ hacia arriba y $\varepsilon_2$ hacia la izquierda.
Con 3 inc\'ognitas se necesitan 3 ecuaciones: \textbf{1 de nodos} (2 nodos $-$ 1) y
\textbf{2 de mallas}.

\medskip
\noindent\textit{Convenci\'on para KVL:} al recorrer una resistencia en el sentido de su
corriente el potencial \emph{baja} ($-IR$); al atravesar una fuente desde $-$ hacia $+$ el
potencial \emph{sube} ($+\varepsilon$).

\subsection*{Ecuaciones de Kirchhoff}

\textbf{KCL --- Nodo $A$:} la corriente $I_1$ llega a $A$ y se reparte en $I_2$ e $I_3$:
\[
    \boxed{I_1 = I_2 + I_3} \tag{N}
\]

\textbf{KVL --- Malla $M_1$} (rama izquierda + rama central), recorrida en sentido horario
$B\to R_5\to\varepsilon_1\to R_1\to A\to R_3\to B$:
\begin{align*}
    -I_1R_5 + \varepsilon_1 - I_1R_1 - I_2R_3 &= 0\\[2pt]
    I_1\,(R_1+R_5) + I_2\,R_3 &= \varepsilon_1
\end{align*}
\[
    \boxed{7\,I_1 + 12\,I_2 = 20} \tag{M1}
\]

\textbf{KVL --- Malla $M_2$} (rama central + rama derecha), recorrida en sentido horario
$A\to R_2\to R_4\to\varepsilon_2\to R_6\to B\to R_3\to A$. Al subir por $R_3$ se avanza
\emph{contra} $I_2$, por lo que el potencial sube ($+I_2R_3$):
\begin{align*}
    -I_3R_2 - I_3R_4 + \varepsilon_2 - I_3R_6 + I_2R_3 &= 0\\[2pt]
    I_3\,(R_2+R_4+R_6) - I_2\,R_3 &= \varepsilon_2
\end{align*}
\[
    \boxed{20\,I_3 - 12\,I_2 = 24} \tag{M2}
\]

\subsection*{Resoluci\'on del sistema}

De \textbf{(M2)} se despeja $I_3$ y, con \textbf{(N)}, se expresa $I_1$ en funci\'on de $I_2$:
\begin{align*}
    I_3 &= \frac{24 + 12\,I_2}{20} = 1{,}2 + 0{,}6\,I_2\\[2pt]
    I_1 &= I_2 + I_3 = 1{,}2 + 1{,}6\,I_2
\end{align*}
Reemplazando en \textbf{(M1)}:
\begin{align*}
    7\,(1{,}2 + 1{,}6\,I_2) + 12\,I_2 &= 20\\[2pt]
    8{,}4 + 23{,}2\,I_2 &= 20
    \;\;\Rightarrow\;\; I_2 = \frac{11{,}6}{23{,}2} = 0{,}5\ [\text{A}]
\end{align*}
Luego $I_3 = 1{,}2 + 0{,}6\,(0{,}5) = 1{,}5\ [\text{A}]$ \ e \ $I_1 = 1{,}2 + 1{,}6\,(0{,}5) = 2{,}0\ [\text{A}]$.

\[
    \boxed{I = I_1 = 2{,}0\ [\text{A}]}
\]
El resultado es \textbf{positivo}: la corriente sube por $R_5$, en el mismo sentido de la flecha.

\subsection*{Verificaci\'on}

\textbf{Diferencia de potencial $V_A - V_B$ por las tres ramas} (debe ser la misma):
\begin{align*}
    \text{rama central:}\quad   V_A - V_B &= I_2\,R_3 = (0{,}5)(12) = 6{,}0\ [\text{V}]\\[2pt]
    \text{rama izquierda:}\quad V_A - V_B &= \varepsilon_1 - I_1\,(R_1+R_5) = 20 - (2{,}0)(7) = 6{,}0\ [\text{V}]\ \checkmark\\[2pt]
    \text{rama derecha:}\quad   V_A - V_B &= I_3\,(R_2+R_4+R_6) - \varepsilon_2 = (1{,}5)(20) - 24 = 6{,}0\ [\text{V}]\ \checkmark
\end{align*}

\textbf{Balance de potencia.} En ambas fuentes la corriente sale por el borne $+$, de modo que
las dos \emph{entregan} energ\'ia:
\begin{align*}
    P_{\text{entregada}} &= \varepsilon_1 I_1 + \varepsilon_2 I_3 = (20)(2{,}0) + (24)(1{,}5) = 76\ [\text{W}]\\[2pt]
    P_{\text{disipada}}  &= I_1^2\,(R_1+R_5) + I_2^2\,R_3 + I_3^2\,(R_2+R_4+R_6)\\
                         &= (2{,}0)^2(7) + (0{,}5)^2(12) + (1{,}5)^2(20) = 28 + 3 + 45 = 76\ [\text{W}]\ \checkmark
\end{align*}

\subsection*{Resumen de resultados}

\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{|c|c|l|}
\hline
\textbf{Variable} & \textbf{Valor} & \textbf{Sentido / observaci\'on} \\
\hline
$I_1 = I$ (rama $R_1$, $\varepsilon_1$, $R_5$) & $2{,}0\ [\text{A}]$ & Sube por $R_5$ (igual a la flecha) \\
$I_2$ (rama $R_3$)                            & $0{,}5\ [\text{A}]$ & De $A$ hacia $B$ \\
$I_3$ (rama $R_2$, $R_4$, $\varepsilon_2$, $R_6$) & $1{,}5\ [\text{A}]$ & De $A$ hacia $B$ por la derecha \\
\hline
$V_A - V_B$ & $6{,}0\ [\text{V}]$ & $V_A > V_B$ \\
$P_{\text{entregada}} = P_{\text{disipada}}$ & $76\ [\text{W}]$ & Balance de potencia \\
\hline
\end{tabular}
\end{center}

\subsection*{Datos}

\[
    \varepsilon = 12\ [\text{V}],\qquad
    R_1 = 20\ [\Omega],\qquad
    R_2 = 20\ [\Omega],\qquad
    R_3 = 30\ [\Omega],\qquad
    R_4 = 8{,}0\ [\Omega]
\]

\subsection*{An\'alisis del circuito}

El circuito \textbf{no requiere mallas}: se resuelve por \textbf{reducci\'on serie-paralelo}.
Partiendo de la fuente, la corriente recorre:

\begin{itemize}
    \item \textbf{\textcolor{colR1}{$R_1$}} en serie, desde la fuente hasta el nodo $b$.
    \item En el nodo $b$ la corriente \textbf{se divide} hacia el nodo $c$ por dos caminos:
          \begin{itemize}
              \item la \textbf{\textcolor{colR2}{$R_2$}} (rama central, $I_2 \downarrow$),
              \item la \textbf{\textcolor{colR3}{$R_3$}} ($b \to$ lado derecho $\to c$, $I_3$).
          \end{itemize}
          Como ambas conectan los mismos nodos $b$ y $c$, est\'an \textbf{en paralelo}:
          $R_2 \parallel R_3$.
    \item En $c$ las corrientes se recombinan y atraviesan \textbf{\textcolor{colR4}{$R_4$}}
          en serie, de regreso a la fuente.
\end{itemize}

\noindent En consecuencia, la resistencia equivalente es
$R_{\text{eq}} = R_1 + (R_2 \parallel R_3) + R_4$.

\subsection*{b) Corrientes en las ramas paralelas}

La ca\'ida de tensi\'on entre los nodos $b$ y $c$ es la misma para $R_2$ y $R_3$, e igual
a la que cae en la resistencia equivalente $R_{23}$:
\[
    V_{bc} = I_1\,R_{23} = (0{,}30)(12) = 3{,}6\ [\text{V}]
\]

Aplicando la ley de Ohm en cada rama:
\[
    I_2 = \frac{V_{bc}}{R_2} = \frac{3{,}6}{20}
        \;\Rightarrow\; \boxed{I_2 = 0{,}18\ [\text{A}]}
    \qquad\qquad
    I_3 = \frac{V_{bc}}{R_3} = \frac{3{,}6}{30}
        \;\Rightarrow\; \boxed{I_3 = 0{,}12\ [\text{A}]}
\]

\noindent\textbf{Verificaci\'on (nodo $b$):}
$\;I_2 + I_3 = 0{,}18 + 0{,}12 = 0{,}30 = I_1$ \checkmark

%----------------------------------------
\subsection*{c) Potencia disipada en cada resistencia}

La potencia disipada en cada resistencia es $P = I^2 R$:

\begin{align*}
    P_1 &= I_1^2\,R_1 = (0{,}30)^2 (20) = 1{,}80\ [\text{W}] \\
    P_2 &= I_2^2\,R_2 = (0{,}18)^2 (20) = 0{,}648\ [\text{W}] \\
    P_3 &= I_3^2\,R_3 = (0{,}12)^2 (30) = 0{,}432\ [\text{W}] \\
    P_4 &= I_1^2\,R_4 = (0{,}30)^2 (8{,}0) = 0{,}72\ [\text{W}]
\end{align*}

\[
    \boxed{P_{\text{total}} = P_1 + P_2 + P_3 + P_4 = 3{,}6\ [\text{W}]}
\]

\noindent\small\textit{Comprobaci\'on (balance de potencia):} la potencia entregada por la
fuente es $P = \varepsilon\,I_1 = (12)(0{,}30) = 3{,}6\ [\text{W}]$, igual a la disipada
por las resistencias. \checkmark
